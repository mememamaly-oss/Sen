    18→class ScanConfig:
    19→    def __init__(self, allow_private=False, max_pages=25, crawl=True, active=True,
    20→                 passive=True, timeout=12.0, seed=None, strategy_path=None,
    21→                 allowed_domains=None, oast=None, browser=True, auth=None,
    22→                 openapi=True, openapi_spec=None):
    23→        self.allow_private = allow_private
    24→        self.max_pages = max_pages
    25→        self.crawl, self.active, self.passive = crawl, active, passive
    26→        self.timeout, self.seed = timeout, seed
    27→        self.strategy_path = strategy_path
    28→        self.allowed_domains = allowed_domains or []
    29→        self.oast = oast          # OastServer | None — enables OOB (blind-XXE) confirmation
    30→        self.browser = browser    # Phase 50: DOM engine (graceful skip when unavailable)
    31→        self.auth = auth          # Phase 51: AuthProfile | None — authenticated scanning
    32→        self.openapi = openapi    # DISC-13: auto-probe well-known OpenAPI/Swagger docs
    33→        self.openapi_spec = openapi_spec   # explicit spec dict/url seed (None = auto only)

    36→class _FingerprintingProxy:
    37→    """Phase 54 transparent proxy: every response the scan receives passes
    38→    through FingerprintCollector.observe() — WAF signatures, tech headers,
    39→    cookie platforms, DOM markers accumulate from traffic the scan already
    40→    produces. No extra requests, no behavior change (pure observer)."""
    41→    def __init__(self, inner, collector: FingerprintCollector):
    42→        self._inner, self._fp = inner, collector
    43→
    44→    def send(self, m):
    45→        r = self._inner.send(m)
    46→        self._fp.observe(r)
    47→        return r
    48→
    49→    def __getattr__(self, name):
    50→        return getattr(self._inner, name)

    53→class SentinelScanner:
    54→    def __init__(self, target_url: str, cfg: ScanConfig | None = None):
    55→        self.cfg = cfg or ScanConfig()
    56→        self.run_id, self.seed = new_run_id(self.cfg.seed)
    57→        self.bus = EventBus(self.run_id)
    58→        self.target = Target(target_url)
    59→        self.scope = ScopeRule(allowed_domains=[self.target.host] + self.cfg.allowed_domains,
    60→                               allow_private=self.cfg.allow_private)
    61→        self.engine = RequestEngine(self.scope, self.bus, timeout=self.cfg.timeout)
    62→        self.verifier = VerificationEngine(self.engine, self.bus, oast=self.cfg.oast)
    63→        self.browser = None       # Phase 50: BrowserEngine (set in run() when enabled+available)
    64→        self.fp = FingerprintCollector()   # Phase 54: passive WAF + tech map
    65→        self.pipeline = FindingPipeline(self.bus)
    66→        self.strategies = StrategyStore(self.cfg.strategy_path)
    67→        self.failures = FailureMemory()
    68→        self.classifier = RepairClassifier()
    69→        self.findings: list[Finding] = []
    70→        self.stats = {"requests": 0, "endpoints": 0, "candidates": 0, "verified": 0, "rejected": 0,
    71→                      "duration": 0.0}
    72→
    73→    # ---------------------------------------------------------------- run
    74→    def run(self) -> dict:
    75→        t0 = time.time()
    76→        self.bus.emit("ScanStarted", target=self.target.url, run=self.run_id,
    77→                      scanner=SCANNER_VERSION, seed=self.seed)
    78→
    79→        # ---- Phase 51: authenticate BEFORE discovery — the session cookie (ENG-05
    80→        # jar) then rides every later request, and the authenticated landing page
    81→        # becomes a second crawl seed (the post-login surface is linked ONLY from
    82→        # post-login pages: /bank/* on Altoro is invisible from the anonymous home).
    83→        # AUTH-05 (Phase 54): token mode — Bearer/JWT/custom-header auth. Validation
    84→        # instead of credential exchange; the header rides engine.default_headers;
    85→        # no self-healing (static tokens cannot be re-minted) — 401/403 mid-scan are
    86→        # counted honestly by the TokenWatchdogProxy instead of silently degrading.
    87→        auth_seeds: list[str] = []
    88→        self.auth_engine = None
    89→        self.token_engine = None
    90→        auth_headers: dict = {}
    91→        if self.cfg.auth is not None:
    92→            if self.cfg.auth.mode == "token":
    93→                self.token_engine = TokenAuthEngine(self.cfg.auth, self.engine, self.bus)
    94→                ok = self.token_engine.login()
    95→                self.stats["authenticated"] = ok
    96→                if ok:
    97→                    self.engine = TokenWatchdogProxy(self.engine, self.token_engine)
    98→                    self.verifier.engine = self.engine
    99→                    auth_headers = {self.cfg.auth.auth_header: self.cfg.auth.header_value()}
   100→                    auth_seeds.append(self.cfg.auth.login_url)
   101→                else:
   102→                    self.bus.emit("ScanContinuesUnauthenticated",
   103→                                  reason=self.token_engine.state.last_error)
   104→            else:
   105→                self.auth_engine = AuthEngine(self.cfg.auth, self.engine, self.bus)
   106→                ok = self.auth_engine.login()
   107→                self.stats["authenticated"] = ok
   108→                if ok:
   109→                    # AUTH-03 watchdog proxy: from here on, session loss anywhere in
   110→                    # the scan (discovery, detectors, verifier, browser) self-heals.
   111→                    self.engine = AuthedEngineProxy(self.engine, self.auth_engine)
   112→                    self.verifier.engine = self.engine
   113→                    auth_seeds.append(self.auth_engine.state.landing_url)
   114→                    # the authenticated crawl genuinely has more surface — grow budget
   115→                    # deterministically (fixed 1.5x, no heuristics)
   116→                    self.cfg.max_pages = int(self.cfg.max_pages * 1.5)
   117→                else:
   118→                    self.bus.emit("ScanContinuesUnauthenticated",
   119→                                  reason=self.auth_engine.state.last_error)
   120→
   121→        # ---- Phase 54: fingerprint everything the scan already fetches. The
   122→        # collector rides a transparent proxy placed OUTERMOST (auth watchdog
   123→        # proxies stay inner), so crawl baselines, detector probes and verify
   124→        # replays ALL pass through observe() — zero extra requests, passive only.
   125→        self.engine = _FingerprintingProxy(self.engine, self.fp)
   126→        self.verifier.engine = self.engine
   127→
   128→        # ---- Phase 4: discovery (extra_seeds carry the authenticated landing;
   129→        # the login page itself is excluded from page-fetches — re-fetching it
   130→        # clobbers the authenticated session cookie with a fresh anonymous one)
   131→        d = DiscoveryEngine(self.engine, self.cfg.max_pages)
   132→        endpoints: list[Endpoint] = [Endpoint(url=self.target.url, source=DiscoverySource.SEED)]
   133→        if self.cfg.crawl:
   134→            found, js, crawl_stats = d.crawl(self.target.url, extra_seeds=auth_seeds,
   135→                                             exclude_urls=[self.cfg.auth.login_url] if self.cfg.auth else None)
   136→            endpoints.extend(e for e in found if e.id != endpoints[0].id)
   137→            robots = d.robots(self.target.url)
   138→            endpoints.extend(e for e in robots if all(e.id != x.id for x in endpoints))
   139→            self.stats["crawl"] = crawl_stats
   140→            self.bus.emit("DiscoveryCompleted", **crawl_stats, js_endpoints=len(js))
   141→        # ---- DISC-13 (Phase 54): the target's OWN OpenAPI/Swagger document is a
   142→        # machine-readable surface map — unlinked admin/legacy/API paths that no
   143→        # crawler can reach are declared there with parameters. Auto-probe the
   144→        # well-known locations (and honor an explicit spec); merged endpoints ride
   145→        # the SAME dedup + scope gates as crawl results.
   146→        if self.cfg.openapi or self.cfg.openapi_spec:
   147→            spec_eps, spec_info = [], None
   148→            if self.cfg.openapi_spec:
   149→                from .discovery import parse_openapi, _spec_info
   150→                if isinstance(self.cfg.openapi_spec, dict):
   151→                    spec_eps = parse_openapi(self.cfg.openapi_spec, self.target.url)
   152→                    spec_info = _spec_info(self.cfg.openapi_spec, "(inline-seed)", len(spec_eps))
   153→            else:
   154→                try:
   155→                    spec_eps, spec_info = d.openapi(self.target.url)
   156→                except Exception:
   157→                    spec_eps, spec_info = [], None
   158→            if spec_eps:
   159→                new_eps = [e for e in spec_eps if all(e.id != x.id for x in endpoints)]
   160→                endpoints.extend(new_eps)
   161→                self.stats["openapi"] = dict(spec_info or {})
   162→                self.stats["openapi"]["new_endpoints"] = len(new_eps)
   163→                self.bus.emit("OpenApiSeeded", **(spec_info or {}), new_endpoints=len(new_eps))
   164→            elif self.cfg.openapi_spec is None:
   165→                self.stats["openapi"] = {"source": None, "paths": 0, "new_endpoints": 0}
   166→        self.stats["endpoints"] = len(endpoints)
   167→
   168→        # ---- Phase 50: browser/DOM engine — render-merge BEFORE planning so active
   169→        # detectors cover runtime-discovered surfaces (JS-built forms, rendered links).
   170→        dom_sinks = list(getattr(d, "dom_sinks", [])) if self.cfg.crawl else []
   171→        if self.cfg.browser:
   172→            from .browser import BrowserEngine, browser_available
   173→            if browser_available():
   174→                self.browser = BrowserEngine(self.engine, timeout=self.cfg.timeout,
   175→                                             extra_headers=auth_headers)
   176→                self.verifier.browser = self.browser
   177→                existing = {e.id: e for e in endpoints}
   178→                # Render budget: script-bearing pages first (runtime forms / DOM logic
   179→                # only exist where scripts run — live-caught: a naive first-N endpoint
   180→                # selection missed the JS-built form page entirely), then dom-sink
   181→                # pages, then remaining GET pages up to the cap.
   182→                seen_t = set()
   183→                render_targets = []
   184→                for u in (list(getattr(d, "script_pages", [])) +
   185→                          [s.url for s in dom_sinks] +
   186→                          [e.url for e in endpoints if (e.method or "GET") == "GET"]):
   187→                    if u and u not in seen_t:
   188→                        seen_t.add(u)
   189→                        render_targets.append(u)
   190→                render_targets = render_targets[:15]
   191→                new_eps = self.browser.render_merge(render_targets, self.target.host, existing)
   192→                endpoints.extend(e for e in new_eps if e.id not in {x.id for x in endpoints})
   193→                self.stats["endpoints"] = len(endpoints)
   194→                self.stats["browser_pages"] = self.browser.pages_rendered
   195→                self.bus.emit("BrowserRenderCompleted", pages=self.browser.pages_rendered,
   196→                              new_endpoints=len(new_eps), dom_sink_candidates=len(dom_sinks))
   197→            else:
   198→                self.bus.emit("BrowserUnavailable",
   199→                              note="playwright/chromium not installed — static fallback active")
   200→
   201→        # ---- Phase 7: plan; Phase 8: detect
   202→        detectors = {d.name: d(self.engine, self.bus, self.strategies) for d in ACTIVE_DETECTORS}
   203→        # Phase 53: OAST-channel detectors (SSRF / blind command injection) receive
   204→        # the armed channel; without one they stay inert (response oracles for these
   205→        # classes are noise — live-observed on the cycling stock endpoint).
   206→        for _d in detectors.values():
   207→            if getattr(_d, "needs_oast", False):
   208→                _d.oast = self.cfg.oast
   209→        vuln_to_detector = {"sql_injection": "sqli", "xss_reflected": "xss",
   210→                            "path_traversal": "traversal", "open_redirect": "redirect", "ssrf": "ssrf",
   211→                            "header_injection": "crlf", "xxe": "xxe",
   212→                            "command_injection": "cmdi", "cors_misconfig": "cors",
   213→                            "ssti": "ssti", "csrf": "csrf", "nosql_injection": "nosql",
   214→                            "jwt_security": "jwt", "host_header_injection": "hostheader"}
   215→        plan = []
   216→        for ep in endpoints:
   217→            plan.extend(plan_for(ep))
   218→        self.bus.emit("ScanPlanReady", planned=len(plan))
   219→
   220→        for ep in endpoints:
   221→            if self.cfg.passive:
   222→                r = self.engine.send(RequestMutation(endpoint=ep, note="passive-baseline"))
   223→                if not r.error:
   224→                    for f in passive_audit(ep, r):
   225→                        self.findings.append(self.verifier.verify(f, ep))
   226→            if not self.cfg.active:
   227→                continue
   228→            for dname, det in detectors.items():
   229→                try:
   230→                    if dname == "sqli":
   231→                        cands = det.candidates(ep)
   232→                        if not cands: continue
   233→                        # PER-PARAM ANALYSIS (live-caught on doLogin: old flat resp_map kept
   234→                        # only the LAST param's responses, silently discarding earlier params)
   235→                        from itertools import groupby
   236→                        for pname, grp in groupby(cands, key=lambda t: t[1].param_name):
   237→                            grp = list(grp)
   238→                            resp_map = {}
   239→                            for strat, m in grp:
   240→                                r = self.engine.send(m)
   241→                                if r.error:
   242→                                    fid = self.failures.record("sqli", r.error, {"url": ep.url})
   243→                                    self.bus.emit("ScannerFailed", scanner="sqli", failure=fid, error=r.error)
   244→                                    continue
   245→                                resp_map[strat] = r
   246→                            param_obj = next((p for p in ep.parameters if p.name == pname), None)
   247→                            if param_obj is None or not resp_map:
   248→                                continue
   249→                            cand = det.analyze((ep, param_obj), resp_map)
   250→                            if cand:
   251→                                cand.run_id = self.run_id
   252→                                self.stats["candidates"] += 1
   253→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   254→                                out_f = self.verifier.verify(cand, ep)
   255→                                self.findings.append(out_f)
   256→                                # Phase 19/25: record VERIFIED outcomes only into strategy store
   257→                                for st in ("quote_error", "boolean_and", "or_redirect", "or_differential", "quote_status"):
   258→                                    self.strategies.record("sqli", st, out_f.status == FindingStatus.VERIFIED,
   259→                                                           context=self.target.host)
   260→                    elif dname == "xxe":
   261→                        # DETECT-08 two-phase flow: candidates() returns recon probes (XML
   262→                        # with form fields, root learned from JS); the recon response names
   263→                        # missing elements ("Store ID could not be parsed"), expand() builds
   264→                        # the entity differential probes, analyze() compares them.
   265→                        cands = det.candidates(ep)
   266→                        for strat, m in cands:
   267→                            r0 = self.engine.send(m)
   268→                            if r0.error or r0.status == 0:
   269→                                continue
   270→                            root = strat.split("|")[1]
   271→                            probes = det.expand(ep, r0, root)
   272→                            if not probes:
   273→                                continue
   274→                            resp_map = {}
   275→                            for s2, m2 in probes:
   276→                                r = self.engine.send(m2)
   277→                                if r.error:
   278→                                    fid = self.failures.record("xxe", r.error, {"url": ep.url})
   279→                                    self.bus.emit("ScannerFailed", scanner="xxe", failure=fid, error=r.error)
   280→                                    continue
   281→                                resp_map[s2] = r
   282→                            if not resp_map:
   283→                                continue
   284→                            cand = det.analyze((ep, None), resp_map)
   285→                            # Phase 49 — OOB fallback: response oracle impossible (noise /
   286→                            # non-deterministic answers) but an OAST listener is armed ->
   287→                            # the target itself becomes the evidence channel (blind XXE).
   288→                            if cand is None and self.cfg.oast is not None:
   289→                                self.bus.emit("OobAttemptStarted", scanner="xxe", url=ep.url)
   290→                                cand = det.oob_attempt(ep, root, r0, self.cfg.oast)
   291→                            if cand:
   292→                                cand.run_id = self.run_id
   293→                                self.stats["candidates"] += 1
   294→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   295→                                out_f = self.verifier.verify(cand, ep)
   296→                                self.findings.append(out_f)
   297→                                strat_name = ("oob_callback" if any(e.kind == "oob_callback" for e in cand.evidence)
   298→                                              else "entity_differential")
   299→                                self.strategies.record("xxe", strat_name,
   300→                                                       out_f.status == FindingStatus.VERIFIED,
   301→                                                       context=self.target.host)
   302→                    elif dname in RUN_STYLE_DETECTORS:
   303→                        # CORS (DETECT-13) + Phase 55: endpoint-driven flows that
   304→                        # drive their own request sequences per endpoint (header
   305→                        # probes, baseline+replay pairs, JSON operator batteries) —
   306→                        # none of them is param-bound, so they cannot ride the
   307→                        # generic loop.
   308→                        cand = det.run(ep)
   309→                        if cand:
   310→                            cand.run_id = self.run_id
   311→                            self.stats["candidates"] += 1
   312→                            self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   313→                            out_f = self.verifier.verify(cand, ep)
   314→                            self.findings.append(out_f)
   315→                            self.strategies.record(dname, "primary",
   316→                                                   out_f.status == FindingStatus.VERIFIED,
   317→                                                   context=self.target.host)
   318→                    else:
   319→                        for m in det.candidates(ep):
   320→                            if isinstance(m, tuple):
   321→                                m = m[1]
   322→                            r = self.engine.send(m)
   323→                            if r.error:
   324→                                fid = self.failures.record(det.name, r.error, {"url": ep.url})
   325→                                self.bus.emit("ScannerFailed", scanner=det.name, failure=fid, error=r.error)
   326→                                continue
   327→                            pair = (ep, next((p for p in ep.parameters if p.name == m.param_name), ep.parameters[0] if ep.parameters else None))
   328→                            if not pair[1]: continue
   329→                            cand = det.analyze(pair, r)
   330→                            if cand:
   331→                                cand.run_id = self.run_id
   332→                                self.stats["candidates"] += 1
   333→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   334→                                out_f = self.verifier.verify(cand, ep)
   335→                                self.findings.append(out_f)
   336→                                self.strategies.record(dname, "primary", out_f.status == FindingStatus.VERIFIED,
   337→                                                       context=self.target.host)
   338→                except Exception as e:
   339→                    fid = self.failures.record(dname, f"{type(e).__name__}: {e}", {"url": ep.url})
   340→                    self.bus.emit("ScannerFailed", scanner=dname, failure=fid, error=str(e)[:200])
   341→
   342→        # ---- Phase 50: DOM XSS execution probes (browser; static sink candidates).
   343→        # search sources probe the extracted param names; hash sources take the
   344→        # payload directly in the fragment; referrer/name sources are a documented
   345→        # limit (cannot be set by plain navigation).
   346→        if self.browser is not None and self.cfg.active and dom_sinks:
   347→            try:
   348→                import hashlib as _hb
   349→                seen_probe = set()
   350→                for sink in dom_sinks:
   351→                    if sink.source == "search":
   352→                        probe_params = [p for p in sink.params[:3] if p]
   353→                    elif sink.source == "hash":
   354→                        probe_params = [""]
   355→                    else:
   356→                        continue
   357→                    for pname in probe_params:
   358→                        pkey = f"{sink.url}|{pname}|{sink.source}"
   359→                        if pkey in seen_probe:
   360→                            continue
   361→                        seen_probe.add(pkey)
   362→                        canary = "z" + _hb.sha1(pkey.encode()).hexdigest()[:9]
   363→                        # canonical page URL (crawl params stripped) — the finding and
   364→                        # its re-probe must be reproducible from a clean URL
   365→                        page_url = sink.url.split("?")[0].split("#")[0]
   366→                        r = self.browser.probe_dom_xss(page_url, pname, canary, source=sink.source)
   367→                        if r and r.get("fired"):
   368→                            cand = Finding(type="xss_dom", severity=Severity.HIGH,
   369→                                           endpoint_url=page_url, parameter=pname, method="GET",
   370→                                           evidence=[Evidence("dom_execution",
   371→                                                      f"source={sink.source} sink={sink.sink} — {r['detail']}; "
   372→                                                      f"static flow: {sink.snippet}")],
   373→                                           status=FindingStatus.CANDIDATE)
   374→                            cand.run_id = self.run_id
   375→                            self.stats["candidates"] += 1
   376→                            self.bus.emit("CandidateCreated", type=cand.type, url=sink.url, param=pname)
   377→                            out_f = self.verifier.verify(cand, None)
   378→                            self.findings.append(out_f)
   379→                            self.strategies.record("dom_xss", sink.sink,
   380→                                                   out_f.status == FindingStatus.VERIFIED,
   381→                                                   context=self.target.host)
   382→                self.stats["browser_probes"] = self.browser.probes_sent
   383→            except Exception as e:
   384→                fid = self.failures.record("dom_xss", f"{type(e).__name__}: {e}", {})
   385→                self.bus.emit("ScannerFailed", scanner="dom_xss", failure=fid, error=str(e)[:200])
   386→        if self.browser is not None:
   387→            try:
   388→                self.browser.close()
   389→            except Exception:
   390→                pass
   391→
   392→        # ---- Phase 12: dedup + correlation + risk
   393→        verified = [f for f in self.findings if f.status == FindingStatus.VERIFIED]
   394→        rejected = [f for f in self.findings if f.status == FindingStatus.REJECTED]
   395→        self.stats.update({"requests": self.engine.requests_sent,
   396→                           "verified": len(verified), "rejected": len(rejected),
   397→                           "duration": round(time.time() - t0, 2)})
   398→        if self.auth_engine is not None:
   399→            self.stats["auth"] = {
   400→                "authenticated": self.auth_engine.state.authenticated,
   401→                "landing": self.auth_engine.state.landing_url,
   402→                "logins": self.auth_engine.state.logins,
   403→                "reauths": self.auth_engine.state.reauths,
   404→                "failed_reauths": self.auth_engine.state.failed_reauths,
   405→            }
   406→        elif self.token_engine is not None:
   407→            # AUTH-05: token mode — no re-auth is possible; rejections (401/403
   408→            # while armed) are the honest degradation signal instead.
   409→            self.stats["auth"] = {"mode": "token",
   410→                                  "authenticated": self.token_engine.state.authenticated,
   411→                                  "header": self.cfg.auth.auth_header,
   412→                                  "rejections": self.token_engine.rejections}
   413→        # Phase 54: WAF + technology map (passive, evidence-backed) and rate-limit
   414→        # resilience counters — scan context in the report, auditable without rerun.
   415→        self.stats["fingerprint"] = self.fp.summary()
   416→        base = self._base_engine()
   417→        self.stats["rate_limit"] = {"limited": base.rate_limited,
   418→                                    "retries": base.rate_limit_retries,
   419→                                    "recovered": base.rate_limit_recovered}
   420→        # Phase 52: OAST channel transparency — which channel served OOB evidence
   421→        # (local loopback listener vs public webhook provider) and how many
   422→        # interactions it recorded. Blind-vuln confirmations are auditable from
   423→        # the report alone, without re-running the scan.
   424→        if self.cfg.oast is not None:
   425→            try:
   426→                self.stats["oast"] = {"channel": getattr(self.cfg.oast, "channel", "local"),
   427→                                      "interactions": len(self.cfg.oast.hits())}
   428→            except Exception:
   429→                self.stats["oast"] = {"channel": getattr(self.cfg.oast, "channel", "local"),
   430→                                      "interactions": -1}
   431→        ded = self.pipeline.dedup(verified)
   432→        final = self.pipeline.correlate(ded)
   433→        for f in final:
   434→            f.confidence = max(f.confidence, 0.5)
   435→        self.findings = final + rejected
   436→        for f in final:
   437→            self.bus.emit("FindingCreated", id=f.id, type=f.type, url=f.endpoint_url,
   438→                          param=f.parameter, risk=self.pipeline.risk(f))
   439→        self.bus.emit("ScanCompleted", **self.stats)
   440→        return self.report()

   442→    def _base_engine(self) -> RequestEngine:
   443→        """Peel auth/fingerprint proxies to the real RequestEngine (counter owner)."""
   444→        e = self.engine
   445→        for _ in range(4):
   446→            inner = getattr(e, "_inner", None)
   447→            if inner is None:
   448→                break
   449→            e = inner
   450→        return e

   458→            "rejected_candidates": [f.to_dict() for f in self.findings
   459→                                    if f.status == FindingStatus.REJECTED],
   460→        }
