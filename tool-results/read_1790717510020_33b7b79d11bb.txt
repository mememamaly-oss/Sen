    24→class ScanConfig:
    25→    def __init__(self, allow_private=False, max_pages=25, crawl=True, active=True,
    26→                 passive=True, timeout=12.0, seed=None, strategy_path=None,
    27→                 allowed_domains=None, oast=None, browser=True, auth=None,
    28→                 openapi=True, openapi_spec=None, authz=None, secrets=True,
    29→                 vault=None, vault_key_file=None):
    30→        self.allow_private = allow_private
    31→        self.max_pages = max_pages
    32→        self.crawl, self.active, self.passive = crawl, active, passive
    33→        self.timeout, self.seed = timeout, seed
    34→        self.strategy_path = strategy_path
    35→        self.allowed_domains = allowed_domains or []
    36→        self.oast = oast          # OastServer | None — enables OOB (blind-XXE) confirmation
    37→        self.browser = browser    # Phase 50: DOM engine (graceful skip when unavailable)
    38→        self.auth = auth          # Phase 51: AuthProfile | None — authenticated scanning
    39→        self.openapi = openapi    # DISC-13: auto-probe well-known OpenAPI/Swagger docs
    40→        self.openapi_spec = openapi_spec   # explicit spec dict/url seed (None = auto only)
    41→        self.authz = authz        # Phase 56: multi-user authorization config (dict | None)
    42→        self.secrets = secrets    # Phase 56: credential-exposure scan of passive baselines
    43→        self.vault = vault                     # Phase 57: vault file path | None
    44→        self.vault_key_file = vault_key_file   # Phase 57: authorized key file path

    47→class _FingerprintingProxy:
    48→    """Phase 54 transparent proxy: every response the scan receives passes
    49→    through FingerprintCollector.observe() — WAF signatures, tech headers,
    50→    cookie platforms, DOM markers accumulate from traffic the scan already
    51→    produces. No extra requests, no behavior change (pure observer)."""
    52→    def __init__(self, inner, collector: FingerprintCollector):
    53→        self._inner, self._fp = inner, collector
    54→
    55→    def send(self, m):
    56→        r = self._inner.send(m)
    57→        self._fp.observe(r)
    58→        return r
    59→
    60→    def __getattr__(self, name):
    61→        return getattr(self._inner, name)

    64→class SentinelScanner:
    65→    def __init__(self, target_url: str, cfg: ScanConfig | None = None):
    66→        self.cfg = cfg or ScanConfig()
    67→        self.run_id, self.seed = new_run_id(self.cfg.seed)
    68→        self.bus = EventBus(self.run_id)
    69→        self.target = Target(target_url)
    70→        self.scope = ScopeRule(allowed_domains=[self.target.host] + self.cfg.allowed_domains,
    71→                               allow_private=self.cfg.allow_private)
    72→        self.engine = RequestEngine(self.scope, self.bus, timeout=self.cfg.timeout)
    73→        self.verifier = VerificationEngine(self.engine, self.bus, oast=self.cfg.oast)
    74→        self.browser = None       # Phase 50: BrowserEngine (set in run() when enabled+available)
    75→        self.fp = FingerprintCollector()   # Phase 54: passive WAF + tech map
    76→        self.pipeline = FindingPipeline(self.bus)
    77→        self.strategies = StrategyStore(self.cfg.strategy_path)
    78→        self.failures = FailureMemory()
    79→        self.classifier = RepairClassifier()
    80→        self.findings: list[Finding] = []
    81→        self.vault = None         # Phase 57: opened Credential Vault (armed capture)
    82→        self.stats = {"requests": 0, "endpoints": 0, "candidates": 0, "verified": 0, "rejected": 0,
    83→                      "duration": 0.0}
    84→
    85→    # ---------------------------------------------------------------- run
    86→    def run(self) -> dict:
    87→        t0 = time.time()
    88→        self.bus.emit("ScanStarted", target=self.target.url, run=self.run_id,
    89→                      scanner=SCANNER_VERSION, seed=self.seed)
    90→
    91→        # ---- Phase 57 (v2.9.0): Credential Vault — opened BEFORE any request
    92→        # so a wrong/unauthorized key fails fast as a configuration error,
    93→        # never mid-scan. Every credential the exposure engine proves is then
    94→        # captured encrypted-at-rest; public artifacts keep masks only.
    95→        if self.cfg.secrets and self.cfg.vault and self.cfg.vault_key_file:
    96→            from .vault import Vault
    97→            self.vault = Vault.open(self.cfg.vault, self.cfg.vault_key_file)
    98→            self.bus.emit("VaultUnlocked", org=self.vault.org,
    99→                          actor=self.vault.key_id)
   100→
   101→        # ---- Phase 51: authenticate BEFORE discovery — the session cookie (ENG-05
   102→        # jar) then rides every later request, and the authenticated landing page
   103→        # becomes a second crawl seed (the post-login surface is linked ONLY from
   104→        # post-login pages: /bank/* on Altoro is invisible from the anonymous home).
   105→        # AUTH-05 (Phase 54): token mode — Bearer/JWT/custom-header auth. Validation
   106→        # instead of credential exchange; the header rides engine.default_headers;
   107→        # no self-healing (static tokens cannot be re-minted) — 401/403 mid-scan are
   108→        # counted honestly by the TokenWatchdogProxy instead of silently degrading.
   109→        auth_seeds: list[str] = []
   110→        self.auth_engine = None
   111→        self.token_engine = None
   112→        auth_headers: dict = {}
   113→        if self.cfg.auth is not None:
   114→            if self.cfg.auth.mode == "token":
   115→                self.token_engine = TokenAuthEngine(self.cfg.auth, self.engine, self.bus)
   116→                ok = self.token_engine.login()
   117→                self.stats["authenticated"] = ok
   118→                if ok:
   119→                    self.engine = TokenWatchdogProxy(self.engine, self.token_engine)
   120→                    self.verifier.engine = self.engine
   121→                    auth_headers = {self.cfg.auth.auth_header: self.cfg.auth.header_value()}
   122→                    auth_seeds.append(self.cfg.auth.login_url)
   123→                else:
   124→                    self.bus.emit("ScanContinuesUnauthenticated",
   125→                                  reason=self.token_engine.state.last_error)
   126→            else:
   127→                self.auth_engine = AuthEngine(self.cfg.auth, self.engine, self.bus)
   128→                ok = self.auth_engine.login()
   129→                self.stats["authenticated"] = ok
   130→                if ok:
   131→                    # AUTH-03 watchdog proxy: from here on, session loss anywhere in
   132→                    # the scan (discovery, detectors, verifier, browser) self-heals.
   133→                    self.engine = AuthedEngineProxy(self.engine, self.auth_engine)
   134→                    self.verifier.engine = self.engine
   135→                    auth_seeds.append(self.auth_engine.state.landing_url)
   136→                    # the authenticated crawl genuinely has more surface — grow budget
   137→                    # deterministically (fixed 1.5x, no heuristics)
   138→                    self.cfg.max_pages = int(self.cfg.max_pages * 1.5)
   139→                else:
   140→                    self.bus.emit("ScanContinuesUnauthenticated",
   141→                                  reason=self.auth_engine.state.last_error)
   142→
   143→        # ---- Phase 54: fingerprint everything the scan already fetches. The
   144→        # collector rides a transparent proxy placed OUTERMOST (auth watchdog
   145→        # proxies stay inner), so crawl baselines, detector probes and verify
   146→        # replays ALL pass through observe() — zero extra requests, passive only.
   147→        self.engine = _FingerprintingProxy(self.engine, self.fp)
   148→        self.verifier.engine = self.engine
   149→        # ---- Phase 56: Coverage Engine — every request the scan sends is
   150→        # attributed to its endpoint (passive baseline / secrets scan / active
   151→        # probe / verification replay). Navigation (crawl) and auth-flow
   152→        # requests are intentionally NOT endpoint testing and are excluded.
   153→        self.coverage = CoverageTracker()
   154→        self.engine = _CoverageProxy(self.engine, self.coverage)
   155→        self.verifier.engine = self.engine
   156→
   157→        # ---- Phase 4: discovery (extra_seeds carry the authenticated landing;
   158→        # the login page itself is excluded from page-fetches — re-fetching it
   159→        # clobbers the authenticated session cookie with a fresh anonymous one)
   160→        d = DiscoveryEngine(self.engine, self.cfg.max_pages)
   161→        endpoints: list[Endpoint] = [Endpoint(url=self.target.url, source=DiscoverySource.SEED)]
   162→        if self.cfg.crawl:
   163→            found, js, crawl_stats = d.crawl(self.target.url, extra_seeds=auth_seeds,
   164→                                             exclude_urls=[self.cfg.auth.login_url] if self.cfg.auth else None)
   165→            endpoints.extend(e for e in found if e.id != endpoints[0].id)
   166→            robots = d.robots(self.target.url)
   167→            endpoints.extend(e for e in robots if all(e.id != x.id for x in endpoints))
   168→            self.stats["crawl"] = crawl_stats
   169→            self.bus.emit("DiscoveryCompleted", **crawl_stats, js_endpoints=len(js))
   170→        # ---- DISC-13 (Phase 54): the target's OWN OpenAPI/Swagger document is a
   171→        # machine-readable surface map — unlinked admin/legacy/API paths that no
   172→        # crawler can reach are declared there with parameters. Auto-probe the
   173→        # well-known locations (and honor an explicit spec); merged endpoints ride
   174→        # the SAME dedup + scope gates as crawl results.
   175→        if self.cfg.openapi or self.cfg.openapi_spec:
   176→            spec_eps, spec_info = [], None
   177→            if self.cfg.openapi_spec:
   178→                from .discovery import parse_openapi, _spec_info
   179→                if isinstance(self.cfg.openapi_spec, dict):
   180→                    spec_eps = parse_openapi(self.cfg.openapi_spec, self.target.url)
   181→                    spec_info = _spec_info(self.cfg.openapi_spec, "(inline-seed)", len(spec_eps))
   182→            else:
   183→                try:
   184→                    spec_eps, spec_info = d.openapi(self.target.url)
   185→                except Exception:
   186→                    spec_eps, spec_info = [], None
   187→            if spec_eps:
   188→                new_eps = [e for e in spec_eps if all(e.id != x.id for x in endpoints)]
   189→                endpoints.extend(new_eps)
   190→                self.stats["openapi"] = dict(spec_info or {})
   191→                self.stats["openapi"]["new_endpoints"] = len(new_eps)
   192→                self.bus.emit("OpenApiSeeded", **(spec_info or {}), new_endpoints=len(new_eps))
   193→            elif self.cfg.openapi_spec is None:
   194→                self.stats["openapi"] = {"source": None, "paths": 0, "new_endpoints": 0}
   195→        self.stats["endpoints"] = len(endpoints)
   196→
   197→        # ---- Phase 50: browser/DOM engine — render-merge BEFORE planning so active
   198→        # detectors cover runtime-discovered surfaces (JS-built forms, rendered links).
   199→        dom_sinks = list(getattr(d, "dom_sinks", [])) if self.cfg.crawl else []
   200→        if self.cfg.browser:
   201→            from .browser import BrowserEngine, browser_available
   202→            if browser_available():
   203→                self.browser = BrowserEngine(self.engine, timeout=self.cfg.timeout,
   204→                                             extra_headers=auth_headers)
   205→                self.verifier.browser = self.browser
   206→                existing = {e.id: e for e in endpoints}
   207→                # Render budget: script-bearing pages first (runtime forms / DOM logic
   208→                # only exist where scripts run — live-caught: a naive first-N endpoint
   209→                # selection missed the JS-built form page entirely), then dom-sink
   210→                # pages, then remaining GET pages up to the cap.
   211→                seen_t = set()
   212→                render_targets = []
   213→                for u in (list(getattr(d, "script_pages", [])) +
   214→                          [s.url for s in dom_sinks] +
   215→                          [e.url for e in endpoints if (e.method or "GET") == "GET"]):
   216→                    if u and u not in seen_t:
   217→                        seen_t.add(u)
   218→                        render_targets.append(u)
   219→                render_targets = render_targets[:15]
   220→                new_eps = self.browser.render_merge(render_targets, self.target.host, existing)
   221→                endpoints.extend(e for e in new_eps if e.id not in {x.id for x in endpoints})
   222→                self.stats["endpoints"] = len(endpoints)
   223→                self.stats["browser_pages"] = self.browser.pages_rendered
   224→                self.bus.emit("BrowserRenderCompleted", pages=self.browser.pages_rendered,
   225→                              new_endpoints=len(new_eps), dom_sink_candidates=len(dom_sinks))
   226→            else:
   227→                self.bus.emit("BrowserUnavailable",
   228→                              note="playwright/chromium not installed — static fallback active")
   229→
   230→        # ---- Phase 56: Smart Testing Engine (value-shape parameter intelligence)
   231→        # + the Coverage Engine's discovery ledger. enrich() only UPGRADES param
   232→        # kinds when the observed value proves a more specific shape (JSON object);
   233→        # every skip it causes is recorded as an auditable coverage reason.
   234→        for ep in endpoints:
   235→            self.coverage.discover(ep)
   236→        self.param_profile = ParamIntel.enrich(endpoints)
   237→        for ep in endpoints:
   238→            for p in ep.parameters:
   239→                reason = ParamIntel.skip_reason(p)
   240→                if reason:
   241→                    self.coverage.skipped(ep, reason)
   242→        self.stats["param_intel"] = self.param_profile
   243→
   244→        # ---- Phase 7: plan; Phase 8: detect
   245→        detectors = {d.name: d(self.engine, self.bus, self.strategies) for d in ACTIVE_DETECTORS}
   246→        # Phase 53: OAST-channel detectors (SSRF / blind command injection) receive
   247→        # the armed channel; without one they stay inert (response oracles for these
   248→        # classes are noise — live-observed on the cycling stock endpoint).
   249→        for _d in detectors.values():
   250→            if getattr(_d, "needs_oast", False):
   251→                _d.oast = self.cfg.oast
   252→        vuln_to_detector = {"sql_injection": "sqli", "xss_reflected": "xss",
   253→                            "path_traversal": "traversal", "open_redirect": "redirect", "ssrf": "ssrf",
   254→                            "header_injection": "crlf", "xxe": "xxe",
   255→                            "command_injection": "cmdi", "cors_misconfig": "cors",
   256→                            "ssti": "ssti", "csrf": "csrf", "nosql_injection": "nosql",
   257→                            "jwt_security": "jwt", "host_header_injection": "hostheader"}
   258→        plan = []
   259→        for ep in endpoints:
   260→            plan.extend(plan_for(ep))
   261→        self.bus.emit("ScanPlanReady", planned=len(plan))
   262→
   263→        for ep in endpoints:
   264→            if self.cfg.passive:
   265→                r = self.engine.send(RequestMutation(endpoint=ep, note="passive-baseline"))
   266→                if r.error:
   267→                    self.coverage.skipped(ep, "transport_error")
   268→                else:
   269→                    # Phase 56 honesty gate: an ANONYMOUS scan that cannot open an
   270→                    # endpoint (401/403, or bounced to a login gate) is an
   271→                    # explicitly-accounted untested surface. With credentials
   272→                    # configured the endpoint WAS tested — a 401 there is the
   273→                    # application's answer (e.g. a login form rejecting wrong
   274→                    # creds), not missing coverage.
   275→                    hop = r.hop_location()
   276→                    if self.cfg.auth is None and (r.status in (401, 403)
   277→                                                  or (hop and "login" in hop.lower())):
   278→                        self.coverage.skipped(ep, "requires_authentication")
   279→                if not r.error:
   280→                    for f in passive_audit(ep, r):
   281→                        self.findings.append(self.verifier.verify(f, ep))
   282→                    # ---- Phase 56: Credential Exposure Engine — secrets scan of
   283→                    # the response the scan ALREADY fetched (zero extra requests;
   284→                    # VR-19 re-fetches only for candidates).
   285→                    if self.cfg.secrets:
   286→                        self.coverage.activity(ep, "secrets", "secrets-passive-scan")
   287→                        capture = (self.vault.capture_callback(self.run_id, ep.url)
   288→                                   if self.vault is not None else None)
   289→                        for sf in SecretsScanner().scan(ep, r, capture=capture):
   290→                            sf.run_id = self.run_id
   291→                            self.stats["candidates"] += 1
   292→                            self.bus.emit("CandidateCreated", type=sf.type, url=ep.url,
   293→                                          param=sf.parameter)
   294→                            self.findings.append(self.verifier.verify(sf, ep))
   295→            if not self.cfg.active:
   296→                continue
   297→            for dname, det in detectors.items():
   298→                try:
   299→                    if dname == "sqli":
   300→                        cands = det.candidates(ep)
   301→                        if not cands: continue
   302→                        # PER-PARAM ANALYSIS (live-caught on doLogin: old flat resp_map kept
   303→                        # only the LAST param's responses, silently discarding earlier params)
   304→                        from itertools import groupby
   305→                        for pname, grp in groupby(cands, key=lambda t: t[1].param_name):
   306→                            grp = list(grp)
   307→                            resp_map = {}
   308→                            for strat, m in grp:
   309→                                r = self.engine.send(m)
   310→                                if r.error:
   311→                                    fid = self.failures.record("sqli", r.error, {"url": ep.url})
   312→                                    self.bus.emit("ScannerFailed", scanner="sqli", failure=fid, error=r.error)
   313→                                    continue
   314→                                resp_map[strat] = r
   315→                            param_obj = next((p for p in ep.parameters if p.name == pname), None)
   316→                            if param_obj is None or not resp_map:
   317→                                continue
   318→                            cand = det.analyze((ep, param_obj), resp_map)
   319→                            if cand:
   320→                                cand.run_id = self.run_id
   321→                                self.stats["candidates"] += 1
   322→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   323→                                out_f = self.verifier.verify(cand, ep)
   324→                                self.findings.append(out_f)
   325→                                # Phase 19/25: record VERIFIED outcomes only into strategy store
   326→                                for st in ("quote_error", "boolean_and", "or_redirect", "or_differential", "quote_status"):
   327→                                    self.strategies.record("sqli", st, out_f.status == FindingStatus.VERIFIED,
   328→                                                           context=self.target.host)
   329→                    elif dname == "xxe":
   330→                        # DETECT-08 two-phase flow: candidates() returns recon probes (XML
   331→                        # with form fields, root learned from JS); the recon response names
   332→                        # missing elements ("Store ID could not be parsed"), expand() builds
   333→                        # the entity differential probes, analyze() compares them.
   334→                        cands = det.candidates(ep)
   335→                        for strat, m in cands:
   336→                            r0 = self.engine.send(m)
   337→                            if r0.error or r0.status == 0:
   338→                                continue
   339→                            root = strat.split("|")[1]
   340→                            probes = det.expand(ep, r0, root)
   341→                            if not probes:
   342→                                continue
   343→                            resp_map = {}
   344→                            for s2, m2 in probes:
   345→                                r = self.engine.send(m2)
   346→                                if r.error:
   347→                                    fid = self.failures.record("xxe", r.error, {"url": ep.url})
   348→                                    self.bus.emit("ScannerFailed", scanner="xxe", failure=fid, error=r.error)
   349→                                    continue
   350→                                resp_map[s2] = r
   351→                            if not resp_map:
   352→                                continue
   353→                            cand = det.analyze((ep, None), resp_map)
   354→                            # Phase 49 — OOB fallback: response oracle impossible (noise /
   355→                            # non-deterministic answers) but an OAST listener is armed ->
   356→                            # the target itself becomes the evidence channel (blind XXE).
   357→                            if cand is None and self.cfg.oast is not None:
   358→                                self.bus.emit("OobAttemptStarted", scanner="xxe", url=ep.url)
   359→                                cand = det.oob_attempt(ep, root, r0, self.cfg.oast)
   360→                            if cand:
   361→                                cand.run_id = self.run_id
   362→                                self.stats["candidates"] += 1
   363→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   364→                                out_f = self.verifier.verify(cand, ep)
   365→                                self.findings.append(out_f)
   366→                                strat_name = ("oob_callback" if any(e.kind == "oob_callback" for e in cand.evidence)
   367→                                              else "entity_differential")
   368→                                self.strategies.record("xxe", strat_name,
   369→                                                       out_f.status == FindingStatus.VERIFIED,
   370→                                                       context=self.target.host)
   371→                    elif dname in RUN_STYLE_DETECTORS:
   372→                        # CORS (DETECT-13) + Phase 55: endpoint-driven flows that
   373→                        # drive their own request sequences per endpoint (header
   374→                        # probes, baseline+replay pairs, JSON operator batteries) —
   375→                        # none of them is param-bound, so they cannot ride the
   376→                        # generic loop.
   377→                        cand = det.run(ep)
   378→                        if cand:
   379→                            cand.run_id = self.run_id
   380→                            self.stats["candidates"] += 1
   381→                            self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   382→                            out_f = self.verifier.verify(cand, ep)
   383→                            self.findings.append(out_f)
   384→                            self.strategies.record(dname, "primary",
   385→                                                   out_f.status == FindingStatus.VERIFIED,
   386→                                                   context=self.target.host)
   387→                    else:
   388→                        for m in det.candidates(ep):
   389→                            if isinstance(m, tuple):
   390→                                m = m[1]
   391→                            r = self.engine.send(m)
   392→                            if r.error:
   393→                                fid = self.failures.record(det.name, r.error, {"url": ep.url})
   394→                                self.bus.emit("ScannerFailed", scanner=det.name, failure=fid, error=r.error)
   395→                                continue
   396→                            pair = (ep, next((p for p in ep.parameters if p.name == m.param_name), ep.parameters[0] if ep.parameters else None))
   397→                            if not pair[1]: continue
   398→                            cand = det.analyze(pair, r)
   399→                            if cand:
   400→                                cand.run_id = self.run_id
   401→                                self.stats["candidates"] += 1
   402→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   403→                                out_f = self.verifier.verify(cand, ep)
   404→                                self.findings.append(out_f)
   405→                                self.strategies.record(dname, "primary", out_f.status == FindingStatus.VERIFIED,
   406→                                                       context=self.target.host)
   407→                except Exception as e:
   408→                    fid = self.failures.record(dname, f"{type(e).__name__}: {e}", {"url": ep.url})
   409→                    self.bus.emit("ScannerFailed", scanner=dname, failure=fid, error=str(e)[:200])
   410→
   411→        # ---- Phase 50: DOM XSS execution probes (browser; static sink candidates).
   412→        # search sources probe the extracted param names; hash sources take the
   413→        # payload directly in the fragment; referrer/name sources are a documented
   414→        # limit (cannot be set by plain navigation).
   415→        if self.browser is not None and self.cfg.active and dom_sinks:
   416→            try:
   417→                import hashlib as _hb
   418→                seen_probe = set()

   584→        """Peel auth/fingerprint proxies to the real RequestEngine (counter owner)."""
   585→        e = self.engine
   586→        for _ in range(4):
   587→            inner = getattr(e, "_inner", None)
   588→            if inner is None:
   589→                break
   590→            e = inner
   591→        return e
