   370→   370→            if r.error or not r.body:
   371→   371→                continue
   372→   372→            forms = self._in_cluster_forms(r.body, url, cluster)
   373→   373→            if forms:
   374→   374→                entry, entry_forms = url, forms
   375→   375→                entry_fields = self._synth_fields(forms[0])
   376→   376→                break
   377→   377→        if not entry:
   378→   378→            return None, "", None, []
   379→   379→        steps: list[WalkStep] = []
   380→   380→        page_url, forms, seen = entry, entry_forms, set()
   381→   381→        limit = max_chain if max_chain is not None else self.cfg.max_steps
   382→   382→
   383→   383→        def _pending():
   384→   384→            """The form the chain would submit next: first UNSEEN in-cluster
   385→   385→            POST form on the current page (None when the chain ended)."""
   386→   386→            return next((f for f in forms if self._path_of(f["action"]) not in seen),
   387→   387→                        None)
   388→   388→
   389→   389→        while len(steps) < limit:
   390→   390→            nxt = _pending()
   391→   391→            if nxt is None:
   392→   392→                return steps, entry, None, entry_fields
   393→   393→            fields = self._synth_fields(nxt, canary=canary,
   394→   394→                                        overrides=option_overrides)
   395→   395→            r = self._post(engine, nxt["action"], fields, "wf-walk")
   396→   396→            seen.add(self._path_of(nxt["action"]))
   397→   397→            step = WalkStep(action=nxt["action"], fields=fields,
   398→   398→                            status=r.status, landing=r.body or "")
   399→   399→            steps.append(step)
   400→   400→            if after_step is not None:
   401→   401→                try:
   402→   402→                    after_step(step, r)
   403→   403→                except Exception:
   404→   404→                    pass
   405→   405→            if r.error or not r.ok:
   406→   406→                return steps, entry, None, entry_fields
   407→   407→            page_url, forms = nxt["action"], self._in_cluster_forms(r.body or "",
   408→   408→                                                                    nxt["action"], cluster)
   409→   409→        return steps, entry, _pending(), entry_fields
   410→   410→
   411→   411→    # -------------------------------------------------- phase 1: learn
   412→   412→    def learn(self, directory: str, members: dict, engine) -> FlowKnowledge:
   413→   413→        k = FlowKnowledge(cluster=directory, members=members)
   414→   414→        steps, entry, pending, entry_fields = self._walk(members, engine)
   415→   415→        if not entry:
   416→   416→            k.drop_reason = "no_anchor"
   417→   417→            return k
   418→   418→        k.entry, k.steps = entry, steps
   419→   419→        if len(steps) < 2:
   420→   420→            k.drop_reason = "chain_too_short"
   421→   421→            return k
   422→   422→        # -- resource linkage: hidden fields that appear in a step's landing
   423→   423→        # form but in no earlier form NOR in the step's own submitted fields —
   424→   424→        # the object ids one step MINTS and a later step consumes. (The step's
   425→   425→        # own fields are registered BEFORE its landing is scanned: the confirm
   426→   426→        # form carries a hidden echo of the amount the create step submitted —
   427→   427→        # that is data continuity, not resource production.)
   428→   428→        known = {n for n, _ in entry_fields}
   429→   429→        for s in steps:
   430→   430→            known |= {n for n, _ in s.fields}
   431→   431→            for form in self._in_cluster_forms(s.landing, s.action, k.cluster):
   432→   432→                for f in form["fields"]:
   433→   433→                    if isinstance(f, dict) and f.get("type") == "hidden" \
   434→   434→                            and f["name"] not in known and not _is_token(f["name"]):
   435→   435→                        k.resource_params[f["name"]] = f.get("value", "")
   436→   436→        if not k.resource_params:
   437→   437→            k.drop_reason = "no_resource_linkage"
   438→   438→            return k
   439→   439→        res_name = next(iter(k.resource_params))
   440→   440→        # -- state oracle: the cluster GET page that echoes the resource value
   441→   441→        origin = self._origin(members)
   442→   442→        for path in sorted(members)[: self.cfg.max_pages]:
   443→   443→            ep = members[path]
   444→   444→            if (ep.method or "GET").upper() != "GET":
   445→   445→                continue
   446→   446→            r = self._get(engine, origin + path, "wf-oracle")
   447→   447→            if not r.error and res_name and any(
   448→   448→                    v and v in (r.body or "") for v in k.resource_params.values()):
   449→   449→                k.oracle_url = ep.url
   450→   450→                break
   451→   451→        if not k.oracle_url:
   452→   452→            k.drop_reason = "no_state_oracle"
   453→   453→            return k
   454→   454→        # -- execution markers: walk the chain AGAIN on the same session,
   455→   455→        # snapshotting the oracle AFTER EVERY STEP, LIVE (a post-hoc snapshot
   456→   456→        # would show the final state everywhere and the delta would be
   457→   457→        # invisible — live-caught: the first v3.0.0 run dropped both flows as
   458→   458→        # no_execution_marker). The producer step is the one whose landing
   459→   459→        # CARRIES the fresh resource id; its snapshot's line is the
   460→   460→        # pre-execution form, the final step's snapshot the executed form.
   461→   461→        snaps: list[str] = []
   462→   462→        steps2, _, _, _ = self._walk(members, engine,
   463→   463→                                     after_step=lambda s, r: snaps.append(
   464→   464→                                         self._get(engine, k.oracle_url,
   465→   465→                                                   "wf-marker").body or ""))
   466→   466→        if not steps2 or not snaps:
   467→   467→            k.drop_reason = "marker_walk_failed"
   468→   468→            return k
   469→   469→        res_val2 = ""
   470→   470→        for i, s in enumerate(steps2):
   471→   471→            for form in self._in_cluster_forms(s.landing, s.action, k.cluster):
   472→   472→                for f in form["fields"]:
   473→   473→                    if isinstance(f, dict) and f.get("name") == res_name \
   474→   474→                            and f.get("value"):
   475→   475→                        res_val2, k.producer_idx = f["value"], i
   476→   476→        if not res_val2 or k.producer_idx < 0 or k.producer_idx >= len(snaps):
   477→   477→            k.drop_reason = "marker_walk_failed"
   478→   478→            return k
   479→   479→        la = (_lines_with(snaps[k.producer_idx], res_val2) or [""])[-1]
   480→   480→        lb = (_lines_with(snaps[-1], res_val2) or [""])[-1]
   481→   481→        added, removed = _word_delta(la, lb)
   482→   482→        k.exec_added = [w for w in added if len(w) > 2]
   483→   483→        k.exec_removed = [w for w in removed if len(w) > 2]
   484→   484→        if not k.exec_added:
   485→   485→            k.drop_reason = "no_execution_marker"
   486→   486→            return k
   487→   487→        # -- privilege-declared options across every form the chain exposes
   488→   488→        seen_opts: list[tuple[str, str, str]] = []
   489→   489→        for s in [WalkStep(action=entry, landing="")] + steps:
   490→   490→            for form in self._in_cluster_forms(s.landing, s.action, k.cluster):
   491→   491→                for triple in _privilege_options([form]):
   492→   492→                    if triple not in seen_opts:
   493→   493→                        seen_opts.append(triple)
   494→   494→        k.priv_options = sorted(seen_opts)[: self.cfg.max_options]
   495→   495→        return k

   498→   498→    def _walk_resource(self, steps, res_name: str, cluster: str) -> str:
   499→   499→        """The resource value a walk's OWN chain minted: the LAST value the
   500→   500→        resource param takes across the walk's landing forms. Every walk mints
   501→   501→        a fresh id — searching the LEARN-time examples instead (the first
   502→   502→        v3.0.0 bug of this class) finds nothing and starves the privilege
   503→   503→        baseline + attack."""
   504→   504→        val = ""
   505→   505→        for s in steps or []:
   506→   506→            for form in self._in_cluster_forms(s.landing, s.action, cluster):
   507→   507→                for f in form["fields"]:
   508→   508→                    if isinstance(f, dict) and f.get("name") == res_name \
   509→   509→                            and f.get("value"):
   510→   510→                        val = f["value"]
   511→   511→        return val

   541→   541→    def _atk_bypass(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   542→   542→        """1. STEP BYPASS — submit the producer step directly on a session that
   543→   543→        never ran its predecessor. Verified iff the oracle gains a NEW
   544→   544→        canary-bearing object line (the object was created anyway)."""
   545→   545→        if k.producer_idx < 1:
   546→   546→            return None, {"verdict": "skipped", "note": "producer is the first step"}
   547→   547→        producer = k.steps[k.producer_idx]
   548→   548→        fields = self._token_refresh(producer.fields, entry_fields)
   549→   549→        pre = self._oracle(engine, k, "wf-atk-bypass-pre")
   550→   550→        r = self._post(engine, producer.action, fields, "wf-atk-bypass")
   551→   551→        post = self._oracle(engine, k, "wf-atk-bypass-post")
   552→   552→        new = [l for l in _lines_with(post, CANARY_CREATE)
   553→   553→               if l not in set(_lines_with(pre, CANARY_CREATE))]
   554→   554→        if not (r.ok and new):
   555→   555→            return None, {"verdict": "no_proof", "status": r.status,
   556→   556→                          "note": "no new canary-bearing object in the state oracle"}
   557→   557→        param = next((n for n, _ in fields if not _is_token(n)), "")
   558→   558→        f = self._finding(
   559→   559→            "business_logic_step_bypass", producer.action, param,
   560→   560→            Evidence("mandatory_step_bypassed",
   561→   561→                     f"step {self._path_of(producer.action)} created an object on a "
   562→   562→                     f"session that never ran its predecessor "
   563→   563→                     f"{self._path_of(k.steps[k.producer_idx - 1].action)} — oracle "
   564→   564→                     f"gained the line '{new[0].strip()[:120]}' — the mandatory "
   565→   565→                     "earlier step is not enforced server-side", r.request_id),
   566→   566→            f"state oracle gained a canary object without the predecessor step: "
   567→   567→            f"'{new[0].strip()[:100]}'")
   568→   568→        return f, {"verdict": "verified", "status": r.status,
   569→   569→                   "line": new[0].strip()[:120]}

   571→   571→    def _atk_out_of_order(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   572→   572→        """2. OUT-OF-ORDER — the FINAL step with a never-created resource id.
   573→   573→        Verified iff the oracle shows that id EXECUTED."""
   574→   574→        if not k.resource_params:
   575→   575→            return None, {"verdict": "skipped", "note": "no resource linkage"}
   576→   576→        final = k.steps[-1]
   577→   577→        res_name = next(iter(k.resource_params))
   578→   578→        fresh_val = _synth_fresh_resource(k.resource_params[res_name])
   579→   579→        fields = self._token_refresh(final.fields, entry_fields)
   580→   580→        fields = [(n, fresh_val if n == res_name
   581→   581→                   else (CANARY_TAMPER if _is_amount(n) else v)) for n, v in fields]
   582→   582→        self._oracle(engine, k, "wf-atk-order-pre")
   583→   583→        r = self._post(engine, final.action, fields, "wf-atk-order")
   584→   584→        post = self._oracle(engine, k, "wf-atk-order-post")
   585→   585→        lines = _lines_with(post, fresh_val)
   586→   586→        if not (r.ok and lines and self._executed(lines[-1], k)):
   587→   587→            return None, {"verdict": "no_proof", "status": r.status,
   588→   588→                          "note": f"synthetic resource {fresh_val} not executed"}
   589→   589→        f = self._finding(
   590→   590→            "business_logic_out_of_order", final.action, res_name,
   591→   591→            Evidence("final_step_without_producer",
   592→   592→                     f"the final step {self._path_of(final.action)} EXECUTED a "
   593→   593→                     f"never-created resource ({res_name}={fresh_val}) straight from "
   594→   594→                     f"POSTed parameters — oracle line '{lines[-1].strip()[:120]}' — "
   595→   595→                     "implicit creation + execution: the producing step was skipped",
   596→   596→                     r.request_id),
   597→   597→            f"oracle shows the synthetic resource executed: '{lines[-1].strip()[:100]}'")
   598→   598→        return f, {"verdict": "verified", "status": r.status, "resource": fresh_val}

   600→   600→    def _atk_tamper(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   601→   601→        """3. VALUE TAMPER — walk the chain legitimately up to the final step,
   602→   602→        then submit the final step with a DIFFERENT amount. Verified iff the
   603→   603→        executed oracle line carries the tampered canary and not the approved
   604→   604→        one."""
   605→   605→        if len(k.steps) < 2 or k.producer_idx < 0:
   606→   606→            return None, {"verdict": "skipped", "note": "chain too short"}
   607→   607→        pre = self._oracle(engine, k, "wf-atk-tamper-pre")
   608→   608→        # live walk up to (excluding) the final step; the pending form is the
   609→   609→        # final step's form with the FRESH resource id and approved canary
   610→   610→        steps, _, pending, _ = self._walk(k.members, engine,
   611→   611→                                          max_chain=len(k.steps) - 1)
   612→   612→        if not steps or pending is None:
   613→   613→            return None, {"verdict": "no_proof", "note": "walk to final step failed"}
   614→   614→        res_name = next(iter(k.resource_params))
   615→   615→        res_val = ""
   616→   616→        for f in pending["fields"]:
   617→   617→            if f.get("name") == res_name:
   618→   618→                res_val = f.get("value", "")
   619→   619→        if not res_val:
   620→   620→            return None, {"verdict": "no_proof",
   621→   621→                          "note": "fresh resource value not observable"}
   622→   622→        mid = self._oracle(engine, k, "wf-atk-tamper-mid")
   623→   623→        if res_val not in mid:
   624→   624→            return None, {"verdict": "no_proof",
   625→   625→                          "note": "created resource not visible in the oracle"}
   626→   626→        # synthesize the final step's fields, then FORCE the amount to the
   627→   627→        # tamper canary — the approved value rides a HIDDEN field, so plain
   628→   628→        # synthesis would carry it verbatim; the tamper is precisely changing it
   629→   629→        fields = [(n, CANARY_TAMPER if _is_amount(n) else v)
   630→   630→                  for n, v in self._synth_fields(pending)]
   631→   631→        r = self._post(engine, pending["action"], fields, "wf-atk-tamper")
   632→   632→        post = self._oracle(engine, k, "wf-atk-tamper-post")
   633→   633→        line = (_lines_with(post, res_val) or [""])[-1]
   634→   634→        if not (r.ok and line and CANARY_TAMPER in line and CANARY_CREATE not in line
   635→   635→                and self._executed(line, k)):
   636→   636→            return None, {"verdict": "no_proof", "status": r.status,
   637→   637→                          "note": "approved value not overridable/executed"}
   638→   638→        param = next((n for n, _ in fields if _is_amount(n)), "")
   639→   639→        f = self._finding(
   640→   640→            "business_logic_value_tamper", pending["action"], param,
   641→   641→            Evidence("post_approval_override",
   642→   642→                     f"the final step executed {res_name}={res_val} with the TAMPERED "
   643→   643→                     f"value ({CANARY_TAMPER}) replacing the approved one "
   644→   644→                     f"({CANARY_CREATE}) — oracle line '{line.strip()[:120]}' — values "
   645→   645→                     "are re-trusted at execution time instead of bound at approval",
   646→   646→                     r.request_id),
   647→   647→            f"executed oracle line carries the tampered value: '{line.strip()[:100]}'")
   648→   648→        return f, {"verdict": "verified", "status": r.status,
   649→   649→                   "line": line.strip()[:120]}

   651→   651→    def _atk_duplicate(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   652→   652→        """4. DUPLICATE EXECUTION — run the full chain, then replay the final
   653→   653→        step once. Verified iff the same resource shows MORE oracle lines
   654→   654→        after the replay than before it."""
   655→   655→        if len(k.steps) < 2:
   656→   656→            return None, {"verdict": "skipped", "note": "chain too short"}
   657→   657→        steps, _, _, _ = self._walk(k.members, engine)
   658→   658→        if not steps or len(steps) < 2:
   659→   659→            return None, {"verdict": "no_proof", "note": "full walk failed"}
   660→   660→        final = steps[-1]
   661→   661→        res_name = next(iter(k.resource_params))
   662→   662→        res_val = ""
   663→   663→        for n, v in final.fields:
   664→   664→            if n == res_name:
   665→   665→                res_val = v
   666→   666→        mid = self._oracle(engine, k, "wf-atk-dup-mid")
   667→   667→        if not res_val or res_val not in mid:
   668→   668→            return None, {"verdict": "no_proof",
   669→   669→                          "note": "executed resource not observable in the oracle"}
   670→   670→        r = self._post(engine, final.action, final.fields, "wf-atk-dup-replay")
   671→   671→        post = self._oracle(engine, k, "wf-atk-dup-post")
   672→   672→        n_mid, n_post = len(_lines_with(mid, res_val)), len(_lines_with(post, res_val))
   673→   673→        if not (r.ok and n_post > n_mid):
   674→   674→            return None, {"verdict": "no_proof", "status": r.status,
   675→   675→                          "note": f"executions {n_mid} -> {n_post}"}
   676→   676→        f = self._finding(
   677→   677→            "business_logic_duplicate_execution", final.action, res_name,
   678→   678→            Evidence("non_idempotent_replay",
   679→   679→                     f"replaying the final step executed {res_name}={res_val} AGAIN: "
   680→   680→                     f"oracle occurrences {n_mid} -> {n_post} — no one-shot consumption "
   681→   681→                     "(missing idempotency guard)", r.request_id),
   682→   682→            f"same resource executed twice: {n_mid} -> {n_post} oracle occurrences")
   683→   683→        return f, {"verdict": "verified", "status": r.status,
   684→   684→                   "executions": f"{n_mid}->{n_post}"}

   686→   686→    def _atk_privilege(self, k: FlowKnowledge, low_engine, entry_fields,
   687→   687→                       low_role: str) -> tuple[Finding | None, dict]:
   688→   688→        """5. PRIVILEGE BOUND — walk the chain under a LOWER-privilege session
   689→   689→        choosing the option the UI labels privileged. Verified iff that
   690→   690→        session's oracle shows the option EXECUTED (baseline: the option is
   691→   691→        functional for the primary role — a control broken for everyone is
   692→   692→        never reported as an escalation)."""
   693→   693→        if not k.priv_options:
   694→   694→            return None, {"verdict": "skipped", "note": "no privilege-declared option"}
   695→   695→        fname, opt, label = k.priv_options[0]
   696→   696→        if not k.priv_baselines.get(opt):
   697→   697→            return None, {"verdict": "skipped",
   698→   698→                          "note": f"option {opt} not functional for the primary role"}
   699→   699→        pre = self._oracle(low_engine, k, "wf-atk-priv-pre")
   700→   700→        steps, _, _, _ = self._walk(k.members, low_engine,
   701→   701→                                    option_overrides={fname: opt})
   702→   702→        if not steps:
   703→   703→            return None, {"verdict": "no_proof", "note": "low-privilege walk failed"}
   704→   704→        mid = self._oracle(low_engine, k, "wf-atk-priv-post")
   705→   705→        res_val = self._walk_resource(steps, next(iter(k.resource_params)), k.cluster)
   706→   706→        if not res_val or res_val in pre or res_val not in mid:
   707→   707→            return None, {"verdict": "no_proof",
   708→   708→                          "note": "low-privilege walk produced no observable object"}
   709→   709→        line = (_lines_with(mid, res_val) or [""])[-1]
   710→   710→        if not (opt in line and self._executed(line, k)):
   711→   711→            return None, {"verdict": "no_proof",
   712→   712→                          "note": f"option {opt} not executed by role={low_role}"}
   713→   713→        f = self._finding(
   714→   714→            "business_logic_privilege_escalation", k.steps[-1].action, fname,
   715→   715→            Evidence("declared_privilege_not_enforced",
   716→   716→                     f"the UI declares option {fname}={opt} '{label}' as privileged, "
   717→   717→                     f"yet a role={low_role} session EXECUTED it — oracle line "
   718→   718→                     f"'{line.strip()[:120]}' — the declared boundary is not enforced "
   719→   719→                     "server-side", ""),
   720→   720→            f"low-privilege session executed the admin-declared option: "
   721→   721→            f"'{line.strip()[:100]}'",
   722→   722→            authz_context=f"attacker=role:{low_role} declared_privilege={opt}")
   723→   723→        return f, {"verdict": "verified", "role": low_role, "option": opt}

   726→   726→    def run(self, endpoints: list, main_engine, scope, bus, auth_profile,
   727→   727→            authz_sessions: dict | None = None, timeout: float = 12.0,
   728→   728→            logic_active: bool = False) -> dict:
   729→   729→        """Phase 58 entry point. Learning always runs (when a form session
   730→   730→        exists); the attack battery only with logic_active=True."""
   731→   731→        stats = {"enabled": True, "logic_active": logic_active,
   732→   732→                 "clusters_tried": 0, "flows_learned": 0, "attacks": 0,
   733→   733→                 "verified": 0, "skipped": [], "requests": 0}
   734→   734→        findings: list[Finding] = []
   735→   735→        flows: list[FlowKnowledge] = []
   736→   736→        if auth_profile is None or getattr(auth_profile, "mode", "form") != "form":
   737→   737→            stats.update(enabled=False, skipped=["no_form_session"])
   738→   738→            return {"findings": findings, "stats": stats, "flows": flows}
   739→   739→        self._main_engine = main_engine
   740→   740→        login_urls = {getattr(auth_profile, "login_url", ""),
   741→   741→                      getattr(auth_profile, "action_url", "")}
   742→   742→        clusters = self._cluster_members(endpoints, login_urls)
   743→   743→        ranked = sorted(((d, m) for d, m in clusters.items() if self._eligible(m)),
   744→   744→                        key=lambda t: (-self._score(t[0], t[1])[0],
   745→   745→                                       -self._score(t[0], t[1])[1], t[0]))
   746→   746→        for directory, members in ranked[: self.cfg.max_clusters]:
   747→   747→            stats["clusters_tried"] += 1
   748→   748→            k = self.learn(directory, members, main_engine)
   749→   749→            if k.drop_reason:
   750→   750→                stats["skipped"].append(f"{directory}:{k.drop_reason}")
   751→   751→                continue
   752→   752→            # privilege baselines under the PRIMARY session: an option broken
   753→   753→            # for everyone is a bug, not an escalation — never reported as one.
   754→   754→            res_name = next(iter(k.resource_params), "")
   755→   755→            for fname, opt, label in k.priv_options:
   756→   756→                pre = self._oracle(main_engine, k, "wf-base-pre")
   757→   757→                bsteps, _, _, _ = self._walk(members, main_engine,
   758→   758→                                             option_overrides={fname: opt})
   759→   759→                post = self._oracle(main_engine, k, "wf-base-post")
   760→   760→                fresh = self._walk_resource(bsteps, res_name, directory)
   761→   761→                line = (_lines_with(post, fresh) or [""])[-1] if fresh else ""
   762→   762→                k.priv_baselines[opt] = bool(fresh and opt in line
   763→   763→                                             and self._executed(line, k))
   764→   764→            stats["flows_learned"] += 1
   765→   765→            flows.append(k)
   766→   766→        if not logic_active:
   767→   767→            stats["requests"] = self.requests
   768→   768→            return {"findings": findings, "stats": stats, "flows": flows}
   769→   769→        # ---- attack battery (explicit opt-in): fresh session per manipulation.
   770→   770→        # The budget is PER FLOW (the five invariants are a flow-sized battery —
   771→   771→        # a global cap let the first flow consume the whole budget and starved
   772→   772→        # the rest: live-caught in benchmark trial 3, duplicate+privilege FN).
   773→   773→        for k in flows:
   774→   774→            flow_attacks = 0
   775→   775→            for name, atk in (("step_bypass", self._atk_bypass),
   776→   776→                              ("out_of_order", self._atk_out_of_order),
   777→   777→                              ("value_tamper", self._atk_tamper),
   778→   778→                              ("duplicate", self._atk_duplicate)):
   779→   779→                if flow_attacks >= self.cfg.max_flow_attacks:
   780→   780→                    stats["skipped"].append(f"{k.cluster}:attack_budget")
   781→   781→                    break
   782→   782→                flow_attacks += 1
   783→   783→                eng = self._fresh_session(scope, bus, auth_profile, timeout)
   784→   784→                if eng is None:
   785→   785→                    stats["skipped"].append(f"{k.cluster}:fresh_login_failed")
   786→   786→                    continue
   787→   787→                entry_fields = []
   788→   788→                r = self._get(eng, k.entry, "wf-atk-entry")
   789→   789→                if not r.error:
   790→   790→                    forms = self._in_cluster_forms(r.body or "", k.entry, k.cluster)
   791→   791→                    entry_fields = self._synth_fields(forms[0]) if forms else []
   792→   792→                if not entry_fields:
   793→   793→                    stats["skipped"].append(f"{k.cluster}:entry_form_lost")
   794→   794→                    continue
   795→   795→                f, detail = atk(k, eng, entry_fields)
   796→   796→                detail.update(attack=name, flow=k.cluster)
   797→   797→                self.log.append(detail)
   798→   798→                stats["attacks"] += 1
   799→   799→                if f:
   800→   800→                    findings.append(f)
   801→   801→                    stats["verified"] += 1
   802→   802→            # privilege: needs a strictly lower-privilege logged-in session
   803→   803→            lows = sorted((s for s in (authz_sessions or {}).values()
   804→   804→                           if s.get("rank", 3) > 0),
   805→   805→                          key=lambda s: s["account"].name)
   806→   806→            if not lows:
   807→   807→                stats["skipped"].append(f"{k.cluster}:no_lower_privilege_session")
   808→   808→            else:
   809→   809→                low = lows[0]
   810→   810→                entry_fields = []
   811→   811→                r = self._get(low["engine"], k.entry, "wf-atk-priv-entry")
   812→   812→                if not r.error:
   813→   813→                    forms = self._in_cluster_forms(r.body or "", k.entry, k.cluster)
   814→   814→                    entry_fields = self._synth_fields(forms[0]) if forms else []
   815→   815→                if entry_fields:
   816→   816→                    f, detail = self._atk_privilege(k, low["engine"], entry_fields,
   817→   817→                                                    low["account"].role)
   818→   818→                    detail.update(attack="privilege", flow=k.cluster)
   819→   819→                    self.log.append(detail)
   820→   820→                    stats["attacks"] += 1
   821→   821→                    if f:
   822→   822→                        findings.append(f)
   823→   823→                        stats["verified"] += 1
   824→   824→                else:
   825→   825→                    stats["skipped"].append(f"{k.cluster}:low_entry_form_lost")
   826→   826→        stats["requests"] = self.requests
   827→   827→        stats["attack_log"] = self.log
   828→   828→        return {"findings": findings, "stats": stats, "flows": flows}
