    49→    49→class _FingerprintingProxy:
    50→    50→    """Phase 54 transparent proxy: every response the scan receives passes
    51→    51→    through FingerprintCollector.observe() — WAF signatures, tech headers,
    52→    52→    cookie platforms, DOM markers accumulate from traffic the scan already
    53→    53→    produces. No extra requests, no behavior change (pure observer)."""
    54→    54→    def __init__(self, inner, collector: FingerprintCollector):
    55→    55→        self._inner, self._fp = inner, collector

    57→    57→    def send(self, m):
    58→    58→        r = self._inner.send(m)
    59→    59→        self._fp.observe(r)
    60→    60→        return r
    61→    61→
    62→    62→    def __getattr__(self, name):
    63→    63→        return getattr(self._inner, name)
