    24→class ScanConfig:

    28→                 openapi=True, openapi_spec=None, authz=None, secrets=True,
    29→                 vault=None, vault_key_file=None, workflow=False, logic_active=False):
    30→        self.allow_private = allow_private
    31→        self.max_pages = max_pages
    32→        self.crawl, self.active, self.passive = crawl, active, passive
    33→        self.timeout, self.seed = timeout, seed
    34→        self.strategy_path = strategy_path

    36→        self.oast = oast          # OastServer | None — enables OOB (blind-XXE) confirmation

    41→        self.authz = authz        # Phase 56: multi-user authorization config (dict | None)
    42→        self.secrets = secrets    # Phase 56: credential-exposure scan of passive baselines
    43→        self.vault = vault                     # Phase 57: vault file path | None
    44→        self.vault_key_file = vault_key_file   # Phase 57: authorized key file path

    46→        self.logic_active = logic_active       # Phase 58: attack battery (--logic-active)

    49→class _FingerprintingProxy:
    50→    """Phase 54 transparent proxy: every response the scan receives passes
    51→    through FingerprintCollector.observe() — WAF signatures, tech headers,
    52→    cookie platforms, DOM markers accumulate from traffic the scan already
    53→    produces. No extra requests, no behavior change (pure observer)."""
    54→    def __init__(self, inner, collector: FingerprintCollector):
    55→        self._inner, self._fp = inner, collector
    56→
    57→    def send(self, m):
    58→        r = self._inner.send(m)
    59→        self._fp.observe(r)
    60→        return r
    61→
    62→    def __getattr__(self, name):
    63→        return getattr(self._inner, name)

    66→class SentinelScanner:
    67→    def __init__(self, target_url: str, cfg: ScanConfig | None = None):
    68→        self.cfg = cfg or ScanConfig()
    69→        self.run_id, self.seed = new_run_id(self.cfg.seed)
    70→        self.bus = EventBus(self.run_id)
    71→        self.target = Target(target_url)
    72→        self.scope = ScopeRule(allowed_domains=[self.target.host] + self.cfg.allowed_domains,
    73→                               allow_private=self.cfg.allow_private)
    74→        self.engine = RequestEngine(self.scope, self.bus, timeout=self.cfg.timeout)
    75→        self.verifier = VerificationEngine(self.engine, self.bus, oast=self.cfg.oast)
    76→        self.browser = None       # Phase 50: BrowserEngine (set in run() when enabled+available)
    77→        self.fp = FingerprintCollector()   # Phase 54: passive WAF + tech map
    78→        self.pipeline = FindingPipeline(self.bus)
    79→        self.strategies = StrategyStore(self.cfg.strategy_path)
    80→        self.failures = FailureMemory()
    81→        self.classifier = RepairClassifier()
    82→        self.findings: list[Finding] = []
    83→        self.vault = None         # Phase 57: opened Credential Vault (armed capture)
    84→        self.stats = {"requests": 0, "endpoints": 0, "candidates": 0, "verified": 0, "rejected": 0,
    85→                      "duration": 0.0}
    86→
    87→    # ---------------------------------------------------------------- run
    88→    def run(self) -> dict:
    89→        t0 = time.time()
    90→        self.bus.emit("ScanStarted", target=self.target.url, run=self.run_id,
    91→                      scanner=SCANNER_VERSION, seed=self.seed)
    92→
    93→        # ---- Phase 57 (v2.9.0): Credential Vault — opened BEFORE any request
    94→        # so a wrong/unauthorized key fails fast as a configuration error,
    95→        # never mid-scan. Every credential the exposure engine proves is then
    96→        # captured encrypted-at-rest; public artifacts keep masks only.
    97→        if self.cfg.secrets and self.cfg.vault and self.cfg.vault_key_file:
    98→            from .vault import Vault
    99→            self.vault = Vault.open(self.cfg.vault, self.cfg.vault_key_file)
   100→            self.bus.emit("VaultUnlocked", org=self.vault.org,
   101→                          actor=self.vault.key_id)
   102→
   103→        # ---- Phase 51: authenticate BEFORE discovery — the session cookie (ENG-05
   104→        # jar) then rides every later request, and the authenticated landing page
   105→        # becomes a second crawl seed (the post-login surface is linked ONLY from
   106→        # post-login pages: /bank/* on Altoro is invisible from the anonymous home).
   107→        # AUTH-05 (Phase 54): token mode — Bearer/JWT/custom-header auth. Validation
   108→        # instead of credential exchange; the header rides engine.default_headers;
   109→        # no self-healing (static tokens cannot be re-minted) — 401/403 mid-scan are
   110→        # counted honestly by the TokenWatchdogProxy instead of silently degrading.
   111→        auth_seeds: list[str] = []
   112→        self.auth_engine = None
   113→        self.token_engine = None
   114→        auth_headers: dict = {}
   115→        if self.cfg.auth is not None:
   116→            if self.cfg.auth.mode == "token":
   117→                self.token_engine = TokenAuthEngine(self.cfg.auth, self.engine, self.bus)
   118→                ok = self.token_engine.login()
   119→                self.stats["authenticated"] = ok
   120→                if ok:
   121→                    self.engine = TokenWatchdogProxy(self.engine, self.token_engine)
   122→                    self.verifier.engine = self.engine
   123→                    auth_headers = {self.cfg.auth.auth_header: self.cfg.auth.header_value()}
   124→                    auth_seeds.append(self.cfg.auth.login_url)
   125→                else:
   126→                    self.bus.emit("ScanContinuesUnauthenticated",
   127→                                  reason=self.token_engine.state.last_error)
   128→            else:
   129→                self.auth_engine = AuthEngine(self.cfg.auth, self.engine, self.bus)
   130→                ok = self.auth_engine.login()
   131→                self.stats["authenticated"] = ok
   132→                if ok:
   133→                    # AUTH-03 watchdog proxy: from here on, session loss anywhere in
   134→                    # the scan (discovery, detectors, verifier, browser) self-heals.
   135→                    self.engine = AuthedEngineProxy(self.engine, self.auth_engine)
   136→                    self.verifier.engine = self.engine
   137→                    auth_seeds.append(self.auth_engine.state.landing_url)
   138→                    # the authenticated crawl genuinely has more surface — grow budget
   139→                    # deterministically (fixed 1.5x, no heuristics)
   140→                    self.cfg.max_pages = int(self.cfg.max_pages * 1.5)
   141→                else:
   142→                    self.bus.emit("ScanContinuesUnauthenticated",
   143→                                  reason=self.auth_engine.state.last_error)
   144→
   145→        # ---- Phase 54: fingerprint everything the scan already fetches. The
   146→        # collector rides a transparent proxy placed OUTERMOST (auth watchdog
   147→        # proxies stay inner), so crawl baselines, detector probes and verify
   148→        # replays ALL pass through observe() — zero extra requests, passive only.
   149→        self.engine = _FingerprintingProxy(self.engine, self.fp)
   150→        self.verifier.engine = self.engine
   151→        # ---- Phase 56: Coverage Engine — every request the scan sends is
   152→        # attributed to its endpoint (passive baseline / secrets scan / active
   153→        # probe / verification replay). Navigation (crawl) and auth-flow
   154→        # requests are intentionally NOT endpoint testing and are excluded.
   155→        self.coverage = CoverageTracker()
   156→        self.engine = _CoverageProxy(self.engine, self.coverage)
   157→        self.verifier.engine = self.engine
   158→
   159→        # ---- Phase 4: discovery (extra_seeds carry the authenticated landing;
   160→        # the login page itself is excluded from page-fetches — re-fetching it
   161→        # clobbers the authenticated session cookie with a fresh anonymous one)
   162→        d = DiscoveryEngine(self.engine, self.cfg.max_pages)
   163→        endpoints: list[Endpoint] = [Endpoint(url=self.target.url, source=DiscoverySource.SEED)]
   164→        if self.cfg.crawl:
   165→            found, js, crawl_stats = d.crawl(self.target.url, extra_seeds=auth_seeds,
   166→                                             exclude_urls=[self.cfg.auth.login_url] if self.cfg.auth else None)
   167→            endpoints.extend(e for e in found if e.id != endpoints[0].id)
   168→            robots = d.robots(self.target.url)
   169→            endpoints.extend(e for e in robots if all(e.id != x.id for x in endpoints))
   170→            self.stats["crawl"] = crawl_stats
   171→            self.bus.emit("DiscoveryCompleted", **crawl_stats, js_endpoints=len(js))
   172→        # ---- DISC-13 (Phase 54): the target's OWN OpenAPI/Swagger document is a
   173→        # machine-readable surface map — unlinked admin/legacy/API paths that no
   174→        # crawler can reach are declared there with parameters. Auto-probe the
   175→        # well-known locations (and honor an explicit spec); merged endpoints ride
   176→        # the SAME dedup + scope gates as crawl results.
   177→        if self.cfg.openapi or self.cfg.openapi_spec:
   178→            spec_eps, spec_info = [], None
   179→            if self.cfg.openapi_spec:
   180→                from .discovery import parse_openapi, _spec_info
   181→                if isinstance(self.cfg.openapi_spec, dict):
   182→                    spec_eps = parse_openapi(self.cfg.openapi_spec, self.target.url)
   183→                    spec_info = _spec_info(self.cfg.openapi_spec, "(inline-seed)", len(spec_eps))
   184→            else:
   185→                try:
   186→                    spec_eps, spec_info = d.openapi(self.target.url)
   187→                except Exception:
   188→                    spec_eps, spec_info = [], None
   189→            if spec_eps:
   190→                new_eps = [e for e in spec_eps if all(e.id != x.id for x in endpoints)]
   191→                endpoints.extend(new_eps)
   192→                self.stats["openapi"] = dict(spec_info or {})
   193→                self.stats["openapi"]["new_endpoints"] = len(new_eps)
   194→                self.bus.emit("OpenApiSeeded", **(spec_info or {}), new_endpoints=len(new_eps))
   195→            elif self.cfg.openapi_spec is None:
   196→                self.stats["openapi"] = {"source": None, "paths": 0, "new_endpoints": 0}
   197→        self.stats["endpoints"] = len(endpoints)
   198→
   199→        # ---- Phase 50: browser/DOM engine — render-merge BEFORE planning so active
   200→        # detectors cover runtime-discovered surfaces (JS-built forms, rendered links).
   201→        dom_sinks = list(getattr(d, "dom_sinks", [])) if self.cfg.crawl else []
   202→        if self.cfg.browser:
   203→            from .browser import BrowserEngine, browser_available
   204→            if browser_available():
   205→                self.browser = BrowserEngine(self.engine, timeout=self.cfg.timeout,
   206→                                             extra_headers=auth_headers)
   207→                self.verifier.browser = self.browser
   208→                existing = {e.id: e for e in endpoints}
   209→                # Render budget: script-bearing pages first (runtime forms / DOM logic
   210→                # only exist where scripts run — live-caught: a naive first-N endpoint
   211→                # selection missed the JS-built form page entirely), then dom-sink
   212→                # pages, then remaining GET pages up to the cap.
   213→                seen_t = set()
   214→                render_targets = []
   215→                for u in (list(getattr(d, "script_pages", [])) +
   216→                          [s.url for s in dom_sinks] +
   217→                          [e.url for e in endpoints if (e.method or "GET") == "GET"]):
   218→                    if u and u not in seen_t:
   219→                        seen_t.add(u)
   220→                        render_targets.append(u)
   221→                render_targets = render_targets[:15]
   222→                new_eps = self.browser.render_merge(render_targets, self.target.host, existing)
   223→                endpoints.extend(e for e in new_eps if e.id not in {x.id for x in endpoints})
   224→                self.stats["endpoints"] = len(endpoints)
   225→                self.stats["browser_pages"] = self.browser.pages_rendered
   226→                self.bus.emit("BrowserRenderCompleted", pages=self.browser.pages_rendered,
   227→                              new_endpoints=len(new_eps), dom_sink_candidates=len(dom_sinks))
   228→            else:
   229→                self.bus.emit("BrowserUnavailable",
   230→                              note="playwright/chromium not installed — static fallback active")
   231→
   232→        # ---- Phase 56: Smart Testing Engine (value-shape parameter intelligence)
   233→        # + the Coverage Engine's discovery ledger. enrich() only UPGRADES param
   234→        # kinds when the observed value proves a more specific shape (JSON object);
   235→        # every skip it causes is recorded as an auditable coverage reason.
   236→        for ep in endpoints:
   237→            self.coverage.discover(ep)
   238→        self.param_profile = ParamIntel.enrich(endpoints)
   239→        for ep in endpoints:
   240→            for p in ep.parameters:
   241→                reason = ParamIntel.skip_reason(p)
   242→                if reason:
   243→                    self.coverage.skipped(ep, reason)
   244→        self.stats["param_intel"] = self.param_profile
   245→
   246→        # ---- Phase 7: plan; Phase 8: detect
   247→        detectors = {d.name: d(self.engine, self.bus, self.strategies) for d in ACTIVE_DETECTORS}
   248→        # Phase 53: OAST-channel detectors (SSRF / blind command injection) receive
   249→        # the armed channel; without one they stay inert (response oracles for these
   250→        # classes are noise — live-observed on the cycling stock endpoint).
   251→        for _d in detectors.values():
   252→            if getattr(_d, "needs_oast", False):
   253→                _d.oast = self.cfg.oast
   254→        vuln_to_detector = {"sql_injection": "sqli", "xss_reflected": "xss",
   255→                            "path_traversal": "traversal", "open_redirect": "redirect", "ssrf": "ssrf",
   256→                            "header_injection": "crlf", "xxe": "xxe",
   257→                            "command_injection": "cmdi", "cors_misconfig": "cors",
   258→                            "ssti": "ssti", "csrf": "csrf", "nosql_injection": "nosql",
   259→                            "jwt_security": "jwt", "host_header_injection": "hostheader"}
   260→        plan = []
   261→        for ep in endpoints:
   262→            plan.extend(plan_for(ep))
   263→        self.bus.emit("ScanPlanReady", planned=len(plan))
   264→
   265→        for ep in endpoints:
   266→            if self.cfg.passive:
   267→                r = self.engine.send(RequestMutation(endpoint=ep, note="passive-baseline"))
   268→                if r.error:
   269→                    self.coverage.skipped(ep, "transport_error")
   270→                else:
   271→                    # Phase 56 honesty gate: an ANONYMOUS scan that cannot open an
   272→                    # endpoint (401/403, or bounced to a login gate) is an
   273→                    # explicitly-accounted untested surface. With credentials
   274→                    # configured the endpoint WAS tested — a 401 there is the
   275→                    # application's answer (e.g. a login form rejecting wrong
   276→                    # creds), not missing coverage.
   277→                    hop = r.hop_location()
   278→                    if self.cfg.auth is None and (r.status in (401, 403)
   279→                                                  or (hop and "login" in hop.lower())):
   280→                        self.coverage.skipped(ep, "requires_authentication")
   281→                if not r.error:
   282→                    for f in passive_audit(ep, r):
   283→                        self.findings.append(self.verifier.verify(f, ep))
   284→                    # ---- Phase 56: Credential Exposure Engine — secrets scan of
   285→                    # the response the scan ALREADY fetched (zero extra requests;
   286→                    # VR-19 re-fetches only for candidates).
   287→                    if self.cfg.secrets:
   288→                        self.coverage.activity(ep, "secrets", "secrets-passive-scan")
   289→                        capture = (self.vault.capture_callback(self.run_id, ep.url)
   290→                                   if self.vault is not None else None)
   291→                        for sf in SecretsScanner().scan(ep, r, capture=capture):
   292→                            sf.run_id = self.run_id
   293→                            self.stats["candidates"] += 1
   294→                            self.bus.emit("CandidateCreated", type=sf.type, url=ep.url,
   295→                                          param=sf.parameter)
   296→                            self.findings.append(self.verifier.verify(sf, ep))
   297→            if not self.cfg.active:
   298→                continue
   299→            for dname, det in detectors.items():
   300→                try:
   301→                    if dname == "sqli":
   302→                        cands = det.candidates(ep)
   303→                        if not cands: continue
   304→                        # PER-PARAM ANALYSIS (live-caught on doLogin: old flat resp_map kept
   305→                        # only the LAST param's responses, silently discarding earlier params)
   306→                        from itertools import groupby
   307→                        for pname, grp in groupby(cands, key=lambda t: t[1].param_name):
   308→                            grp = list(grp)
   309→                            resp_map = {}
   310→                            for strat, m in grp:
   311→                                r = self.engine.send(m)
   312→                                if r.error:
   313→                                    fid = self.failures.record("sqli", r.error, {"url": ep.url})
   314→                                    self.bus.emit("ScannerFailed", scanner="sqli", failure=fid, error=r.error)
   315→                                    continue
   316→                                resp_map[strat] = r
   317→                            param_obj = next((p for p in ep.parameters if p.name == pname), None)
   318→                            if param_obj is None or not resp_map:
   319→                                continue
   320→                            cand = det.analyze((ep, param_obj), resp_map)
   321→                            if cand:
   322→                                cand.run_id = self.run_id
   323→                                self.stats["candidates"] += 1
   324→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   325→                                out_f = self.verifier.verify(cand, ep)
   326→                                self.findings.append(out_f)
   327→                                # Phase 19/25: record VERIFIED outcomes only into strategy store
   328→                                for st in ("quote_error", "boolean_and", "or_redirect", "or_differential", "quote_status"):
   329→                                    self.strategies.record("sqli", st, out_f.status == FindingStatus.VERIFIED,
   330→                                                           context=self.target.host)
   331→                    elif dname == "xxe":
   332→                        # DETECT-08 two-phase flow: candidates() returns recon probes (XML
   333→                        # with form fields, root learned from JS); the recon response names
   334→                        # missing elements ("Store ID could not be parsed"), expand() builds
   335→                        # the entity differential probes, analyze() compares them.
   336→                        cands = det.candidates(ep)
   337→                        for strat, m in cands:
   338→                            r0 = self.engine.send(m)
   339→                            if r0.error or r0.status == 0:
   340→                                continue
   341→                            root = strat.split("|")[1]
   342→                            probes = det.expand(ep, r0, root)
   343→                            if not probes:
   344→                                continue
   345→                            resp_map = {}
   346→                            for s2, m2 in probes:
   347→                                r = self.engine.send(m2)
   348→                                if r.error:
   349→                                    fid = self.failures.record("xxe", r.error, {"url": ep.url})
   350→                                    self.bus.emit("ScannerFailed", scanner="xxe", failure=fid, error=r.error)
   351→                                    continue
   352→                                resp_map[s2] = r
   353→                            if not resp_map:
   354→                                continue
   355→                            cand = det.analyze((ep, None), resp_map)
   356→                            # Phase 49 — OOB fallback: response oracle impossible (noise /
   357→                            # non-deterministic answers) but an OAST listener is armed ->
   358→                            # the target itself becomes the evidence channel (blind XXE).
   359→                            if cand is None and self.cfg.oast is not None:
   360→                                self.bus.emit("OobAttemptStarted", scanner="xxe", url=ep.url)
   361→                                cand = det.oob_attempt(ep, root, r0, self.cfg.oast)
   362→                            if cand:
   363→                                cand.run_id = self.run_id
   364→                                self.stats["candidates"] += 1
   365→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   366→                                out_f = self.verifier.verify(cand, ep)
   367→                                self.findings.append(out_f)
   368→                                strat_name = ("oob_callback" if any(e.kind == "oob_callback" for e in cand.evidence)
   369→                                              else "entity_differential")
   370→                                self.strategies.record("xxe", strat_name,
   371→                                                       out_f.status == FindingStatus.VERIFIED,
   372→                                                       context=self.target.host)
   373→                    elif dname in RUN_STYLE_DETECTORS:
   374→                        # CORS (DETECT-13) + Phase 55: endpoint-driven flows that
   375→                        # drive their own request sequences per endpoint (header
   376→                        # probes, baseline+replay pairs, JSON operator batteries) —
   377→                        # none of them is param-bound, so they cannot ride the
   378→                        # generic loop.
   379→                        cand = det.run(ep)
   380→                        if cand:
   381→                            cand.run_id = self.run_id
   382→                            self.stats["candidates"] += 1
   383→                            self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   384→                            out_f = self.verifier.verify(cand, ep)
   385→                            self.findings.append(out_f)
   386→                            self.strategies.record(dname, "primary",
   387→                                                   out_f.status == FindingStatus.VERIFIED,
   388→                                                   context=self.target.host)
   389→                    else:
   390→                        for m in det.candidates(ep):
   391→                            if isinstance(m, tuple):
   392→                                m = m[1]
   393→                            r = self.engine.send(m)
   394→                            if r.error:
   395→                                fid = self.failures.record(det.name, r.error, {"url": ep.url})
   396→                                self.bus.emit("ScannerFailed", scanner=det.name, failure=fid, error=r.error)
   397→                                continue
   398→                            pair = (ep, next((p for p in ep.parameters if p.name == m.param_name), ep.parameters[0] if ep.parameters else None))
   399→                            if not pair[1]: continue
   400→                            cand = det.analyze(pair, r)
   401→                            if cand:
   402→                                cand.run_id = self.run_id
   403→                                self.stats["candidates"] += 1
   404→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   405→                                out_f = self.verifier.verify(cand, ep)
   406→                                self.findings.append(out_f)
   407→                                self.strategies.record(dname, "primary", out_f.status == FindingStatus.VERIFIED,
   408→                                                       context=self.target.host)
   409→                except Exception as e:
   410→                    fid = self.failures.record(dname, f"{type(e).__name__}: {e}", {"url": ep.url})
   411→                    self.bus.emit("ScannerFailed", scanner=dname, failure=fid, error=str(e)[:200])
   412→
   413→        # ---- Phase 50: DOM XSS execution probes (browser; static sink candidates).
   414→        # search sources probe the extracted param names; hash sources take the
   415→        # payload directly in the fragment; referrer/name sources are a documented
   416→        # limit (cannot be set by plain navigation).

   631→        """Peel auth/fingerprint proxies to the real RequestEngine (counter owner)."""

   636→                break
   637→            e = inner
