     1→"""SENTINEL v3.0.0 — Workflow / Business-Logic Intelligence Engine (Phase 58).
     2→
     3→Everything SENTINEL tested before v3.0.0 lives at the REQUEST level: one
     4→endpoint, one parameter, one response. This engine moves the frame of
     5→reference to the PROCESS level — the ordered, stateful sequences an
     6→application uses to do real work:
     7→
     8→    login -> select account -> enter amount -> create transfer -> confirm
     9→
    10→It learns those sequences from what the scan already discovered, then asks
    11→the five questions a human business-logic pentester asks (and a per-parameter
    12→scanner structurally cannot):
    13→
    14→  1. STEP BYPASS            can a later step run without its mandatory
    15→                            predecessor?            (create without select)
    16→  2. OUT-OF-ORDER           can the FINAL step run without the step that
    17→                            produces its input?     (confirm a transfer that
    18→                                                     was never created)
    19→  3. VALUE TAMPER           can an already-approved value be changed at
    20→                            execution time?         (amount swapped between
    21→                                                     create and confirm)
    22→  4. DUPLICATE EXECUTION    does replaying the final step execute the
    23→                            operation again?        (no idempotency)
    24→  5. PRIVILEGE BOUND        can a lower-privilege session execute a step
    25→                            the UI itself declares privileged?
    26→                            ("admin only" wire mode)
    27→
    28→LEARN (safe, --workflow): a guided walk submits each form of a flow cluster
    29→once with small canary values, following the chain the application itself
    30→lays out (each landing page carries the next form), and observes which GET
    31→page echoes the produced resource identifiers — the STATE ORACLE. The
    32→word-level diff of a resource's oracle line before and after the final step
    33→(e.g. PENDING -> EXECUTED) is the EXECUTION MARKER vocabulary.
    34→
    35→ATTACK (opt-in, --logic-active): every manipulation runs on a FRESH isolated
    36→session with canary values, and is reported ONLY when the state oracle PROVES
    37→the outcome — a new canary-bearing object, a second execution of the same
    38→resource, or an executed line carrying the tampered canary. A server that
    39→merely fails to reject is NOT a finding; the disallowed thing must actually
    40→happen. Unprovable manipulations (no oracle, no marker, no second role) are
    41→skipped loudly in the stats, never guessed.
    42→
    43→Safety contract: bounded budgets everywhere (clusters, steps, options,
    44→attacks), login/logout-shaped surfaces never submitted, fresh sessions per
    45→manipulation, canaries on every state-changing field, and the whole attack
    46→battery behind an explicit flag.
    47→"""

    83→# ------------------------------------------------------------------ form parsing
    84→class _FlowFormParser(HTMLParser):
    85→    """Form parser that understands <select>/<option> (the crawler's parser
    86→    records plain inputs only — option LABELS are where the UI declares
    87→    privilege boundaries, so this engine parses pages itself)."""
    88→
    89→    def __init__(self):
    90→        super().__init__(convert_charrefs=True)
    91→        self.forms: list[dict] = []
    92→        self._f: dict | None = None
    93→        self._sel: dict | None = None
    94→        self._opt: list | None = None
    95→
    96→    def handle_starttag(self, tag, attrs):
    97→        a = dict(attrs)
    98→        if tag == "form":
    99→            self._f = {"action": a.get("action", ""),
   100→                       "method": (a.get("method") or "GET").upper(), "fields": []}
   101→        elif tag == "select" and self._f is not None and a.get("name"):
   102→            self._sel = {"name": a["name"], "options": []}
   103→            self._f["fields"].append(self._sel)
   104→        elif tag == "option" and self._sel is not None:
   105→            self._opt = [a.get("value", ""), a.get("label") or ""]
   106→            self._sel["options"].append(self._opt)
   107→        elif tag == "input" and self._f is not None:
   108→            if a.get("type") not in ("submit", "button", "reset") and a.get("name"):
   109→                self._f["fields"].append({"name": a["name"], "value": a.get("value", ""),
   110→                                          "type": a.get("type", "text")})
   111→        elif tag == "textarea" and self._f is not None and a.get("name"):
   112→            self._f["fields"].append({"name": a["name"], "value": "", "type": "textarea"})
   113→
   114→    def handle_data(self, data):
   115→        # option labels are element TEXT, not attributes — the privilege
   116→        # declaration ("admin only") lives there
   117→        if self._opt is not None and not self._opt[1] and data.strip():
   118→            self._opt[1] = data.strip()
   119→
   120→    def handle_endtag(self, tag):
   121→        if tag == "option":
   122→            self._opt = None
   123→        elif tag == "select":
   124→            self._sel = None
   125→        elif tag == "form" and self._f is not None:
   126→            if self._f["fields"]:
   127→                self.forms.append(self._f)
   128→            self._f = None
   129→
   130→
   131→def parse_forms(html: str) -> list[dict]:
   132→    p = _FlowFormParser()
   133→    try:
   134→        p.feed(html or "")
   135→    except Exception:
   136→        return []
   137→    return p.forms

   140→def _is_token(name: str) -> bool:
   141→    return any(w in (name or "").lower() for w in TOKEN_WORDS)
   142→
   143→
   144→def _is_amount(name: str) -> bool:
   145→    return any(w in (name or "").lower() for w in AMOUNT_WORDS)

   147→
   148→def _privilege_options(forms: list[dict]) -> list[tuple[str, str, str]]:
   149→    """(field_name, option_value, label) for options whose LABEL declares a
   150→    privileged boundary — the UI itself saying 'only admins get this'."""
   151→    out = []
   152→    for form in forms:
   153→        for f in form["fields"]:
   154→            if isinstance(f, dict) and f.get("options"):
   155→                for val, label in f["options"]:
   156→                    if any(w in (label or "").lower() for w in PRIV_WORDS):
   157→                        triple = (f["name"], val, label)
   158→                        if triple not in out:
   159→                            out.append(triple)
   160→    return out

   163→# ------------------------------------------------------------------ helpers
   164→def _strip_tags(line: str) -> str:
   165→    return re.sub(r"<[^>]+>", " ", line or "")
   166→
   167→
   168→def _lines_with(body: str, value: str) -> list[str]:
   169→    """Tag-stripped record units containing the exact value. Units split on
   170→    newlines AND on list-item/table-row boundaries (two executions of one
   171→    resource appear as two units, whether or not the HTML has newlines)."""
   172→    if not value:
   173→        return []
   174→    units = re.split(r"(?=</?(?:li|tr)\b)|\n", body or "")
   175→    return [_strip_tags(u) for u in units if value in u]
   176→
   177→
   178→def _word_delta(a: str, b: str) -> tuple[list, list]:
   179→    wa, wb = (a or "").split(), (b or "").split()
   180→    return ([w for w in wb if w not in wa], [w for w in wa if w not in wb])

   182→
   183→def _synth_fresh_resource(value: str) -> str:
   184→    """A resource value with the observed SHAPE but never produced by the app:
   185→    'TRF-0007' -> 'TRF-99001' (prefix preserved, id unreachable)."""
   186→    m = re.match(r"^(.*?)(\d+)$", value or "")
   187→    if m and m.group(1):
   188→        return m.group(1) + FRESH_RES_SUFFIX
   189→    return (value or "res") + "-" + FRESH_RES_SUFFIX

   218→# ------------------------------------------------------------------ the engine
   219→class WorkflowEngine:
   220→    """Learn multi-step flows, then test the five business-logic invariants."""
   221→
   222→    def __init__(self, cfg: WorkflowConfig | None = None):
   223→        self.cfg = cfg or WorkflowConfig()
   224→        self.requests = 0            # requests on PRIVATE engines (added to stats)
   225→        self.log: list[dict] = []    # auditable trail of every manipulation
   226→
   227→    # -------------------------------------------------- transport helpers
   228→    def _send(self, engine, m):
   229→        before = engine.requests_sent
   230→        r = engine.send(m)
   231→        if engine is not getattr(self, "_main_engine", None):
   232→            self.requests += engine.requests_sent - before
   233→        return r
   234→
   235→    def _get(self, engine, url, note="wf-get"):
   236→        return self._send(engine, RequestMutation(
   237→            endpoint=Endpoint(url=url, method="GET"), note=note))
   238→
   239→    def _post(self, engine, url, fields, note="wf-post"):
   240→        from .core import Parameter
   241→        params = [Parameter(n, ParamLocation.BODY, v).classify() for n, v in fields]
   242→        return self._send(engine, RequestMutation(
   243→            endpoint=Endpoint(url=url, method="POST", parameters=params), note=note))

   245→    # -------------------------------------------------- form synthesis
   246→    @staticmethod
   247→    def _synth_fields(form: dict, canary: str = CANARY_CREATE,
   248→                      overrides: dict | None = None) -> list:
   249→        """Deterministic values for one form: tokens and hidden fields carried
   250→        verbatim (they ARE the chain), amount-ish fields get the canary,
   251→        selects take their first option unless overridden."""
   252→        out, overrides = [], dict(overrides or {})
   253→        for f in form["fields"]:
   254→            name = f["name"]
   255→            if name in overrides:
   256→                out.append((name, overrides.pop(name)))
   257→            elif isinstance(f, dict) and f.get("options") is not None:
   258→                out.append((name, f["options"][0][0] if f["options"] else ""))
   259→            elif _is_token(name) or f.get("type") == "hidden":
   260→                out.append((name, f.get("value", "")))
   261→            elif _is_amount(name):
   262→                out.append((name, canary))
   263→            else:
   264→                out.append((name, f.get("value") or "1"))
   265→        return out
   266→
   267→    def _token_refresh(self, fields: list, fresh_token_fields: list) -> list:
   268→        """Swap token-kind values for the FRESH session's tokens."""
   269→        fresh = {n: v for n, v in fresh_token_fields if _is_token(n)}
   270→        return [(n, fresh.get(n, v)) if _is_token(n) else (n, v) for n, v in fields]

   272→    # -------------------------------------------------- clustering
   273→    @staticmethod
   274→    def _path_of(url: str) -> str:
   275→        return urllib.parse.urlsplit(url).path or ""
   276→
   277→    def _cluster_members(self, endpoints: list, login_urls: set) -> dict[str, dict]:
   278→        """Group endpoints by directory path. Root, assets, login/logout
   279→        surfaces and the configured auth flow are never flow members."""
   280→        clusters: dict[str, dict] = {}
   281→        for ep in sorted(endpoints, key=lambda e: e.url):
   282→            path = self._path_of(ep.url)
   283→            if not path or path == "/" or ASSET_RE.search(path):
   284→                continue
   285→            if SESSION_KILLER_RE.search(path):
   286→                continue
   287→            if any(u and (self._path_of(u) == path
   288→                          or u.split("?")[0].rstrip("/") == ep.url.split("?")[0].rstrip("/"))
   289→                   for u in login_urls):
   290→                continue
   291→            directory = path.rsplit("/", 1)[0] or "/"
   292→            if directory == "/":
   293→                continue          # flat single-segment surfaces are not flows
   294→            clusters.setdefault(directory, {}).setdefault(path, ep)
   295→        return clusters
   296→
   297→    @staticmethod
   298→    def _eligible(members: dict) -> bool:
   299→        """Fast pre-filter only: a cluster needs at least one crawl-known POST
   300→        (an anchor form's action) and one GET page. The REAL filter is the
   301→        guided walk itself — chain steps two and three levels deep (create,
   302→        confirm) live behind POST landings a GET crawler can never see, so
   303→        requiring them pre-discovered would make the engine structurally
   304→        blind to exactly the flows it exists to learn."""
   305→        posts = [ep for ep in members.values() if (ep.method or "GET").upper() != "GET"]
   306→        gets = [ep for ep in members.values() if (ep.method or "GET").upper() == "GET"]
   307→        return len(posts) >= 1 and len(gets) >= 1 and len(members) >= 2
   308→
   309→    @staticmethod
   310→    def _score(directory: str, members: dict) -> tuple:
   311→        """Flow-worded directories strongly preferred (a 30-endpoint flat test
   312→        directory must not crowd out a 5-endpoint transfer wizard), then size."""
   313→        flowish = 1 if any(w in directory.lower() for w in FLOW_WORDS) else 0
   314→        return (flowish * 10 + len(members), len(members))
   315→
   316→    # -------------------------------------------------- guided walk
   317→    def _origin(self, members: dict) -> str:
   318→        u = urllib.parse.urlsplit(next(iter(members.values())).url)
   319→        return f"{u.scheme}://{u.netloc}"
   320→
   321→    def _in_cluster_forms(self, html: str, page_url: str, cluster: str) -> list[dict]:
   322→        """POST forms whose action lives UNDER the cluster directory — whether
   323→        or not the crawler ever saw that target before. Deep chain steps
   324→        (create, confirm) exist only behind POST landings; requiring crawl
   325→        membership here would hide them (live-caught in the first v3.0.0
   326→        benchmark: 0 flows learned, 5 FN, while the safe twin stayed silent —
   327→        the honest-negative discipline worked, the discovery did not)."""
   328→        out = []
   329→        for f in parse_forms(html):
   330→            if f["method"] != "POST":
   331→                continue
   332→            action = urllib.parse.urljoin(page_url, f["action"] or page_url)
   333→            path = self._path_of(action)
   334→            if (path == cluster or path.startswith(cluster.rstrip("/") + "/")) \
   335→                    and not SESSION_KILLER_RE.search(action):
   336→                out.append(dict(f, action=action))
   337→        return out
   338→
   339→    def _walk(self, members: dict, engine, option_overrides: dict | None = None,
   340→              canary: str = CANARY_CREATE, max_chain: int | None = None,
   341→              after_step=None):
   342→        """The guided walk: fetch cluster GET pages to find the ANCHOR (a GET
   343→        page carrying an in-cluster POST form), submit that form, then follow
   344→        the chain — each landing page carries the next form. `after_step(step,
   345→        response)` fires after every submitted step (the marker learner
   346→        snapshots the state oracle BETWEEN steps — post-hoc snapshots all show
   347→        the final state and the execution delta becomes invisible). Returns
   348→        (steps, entry_url, pending_form, entry_fields) where pending_form is
   349→        the form the chain WOULD submit next (None when the chain ended)."""
   350→        origin = self._origin(members)
   351→        cluster = self._path_of(next(iter(members.values())).url).rsplit("/", 1)[0]
   352→        gets = sorted(((p, ep) for p, ep in members.items()
   353→                       if (ep.method or "GET").upper() == "GET"),
   354→                      key=lambda kv: kv[0])[: self.cfg.max_pages]
   355→        entry, entry_forms, entry_fields = "", [], []
   356→        for path, gep in gets:
   357→            # ENG-16 (live-caught on ginandjuice.shop): the anchor fetch used
   358→            # the BARE path, but query-param pages (/catalog/product?productId=1)
   359→            # only render their form for a valid query — a param-less GET
   360→            # returned an error page with no forms and the whole cluster was
   361→            # dropped as "no_anchor". The endpoint's QUERY parameters carry
   362→            # the discovered values (DISC-07): rebuild the URL with them.
   363→            url = origin + path
   364→            q = "&".join(f"{urllib.parse.quote(p.name)}={urllib.parse.quote(p.value or '')}"
   365→                         for p in (gep.parameters or [])
   366→                         if p.location == ParamLocation.QUERY and p.value)
   367→            if q:
   368→                url += "?" + q
   369→            r = self._get(engine, url, "wf-anchor")
   370→            if r.error or not r.body:
   371→                continue
   372→            forms = self._in_cluster_forms(r.body, url, cluster)
   373→            if forms:
   374→                entry, entry_forms = url, forms
   375→                entry_fields = self._synth_fields(forms[0])
   376→                break
   377→        if not entry:
   378→            return None, "", None, []
   379→        steps: list[WalkStep] = []
   380→        page_url, forms, seen = entry, entry_forms, set()
   381→        limit = max_chain if max_chain is not None else self.cfg.max_steps
   382→
   383→        def _pending():
   384→            """The form the chain would submit next: first UNSEEN in-cluster
   385→            POST form on the current page (None when the chain ended)."""
   386→            return next((f for f in forms if self._path_of(f["action"]) not in seen),
   387→                        None)
   388→
   389→        while len(steps) < limit:
   390→            nxt = _pending()
   391→            if nxt is None:
   392→                return steps, entry, None, entry_fields
   393→            fields = self._synth_fields(nxt, canary=canary,
   394→                                        overrides=option_overrides)
   395→            r = self._post(engine, nxt["action"], fields, "wf-walk")
   396→            seen.add(self._path_of(nxt["action"]))
   397→            step = WalkStep(action=nxt["action"], fields=fields,
   398→                            status=r.status, landing=r.body or "")
   399→            steps.append(step)
   400→            if after_step is not None:
   401→                try:
   402→                    after_step(step, r)
   403→                except Exception:
   404→                    pass
   405→            if r.error or not r.ok:
   406→                return steps, entry, None, entry_fields
   407→            page_url, forms = nxt["action"], self._in_cluster_forms(r.body or "",
   408→                                                                    nxt["action"], cluster)
   409→        return steps, entry, _pending(), entry_fields
   410→
   411→    # -------------------------------------------------- phase 1: learn
   412→    def learn(self, directory: str, members: dict, engine) -> FlowKnowledge:
   413→        k = FlowKnowledge(cluster=directory, members=members)
   414→        steps, entry, pending, entry_fields = self._walk(members, engine)
   415→        if not entry:
   416→            k.drop_reason = "no_anchor"
   417→            return k
   418→        k.entry, k.steps = entry, steps
   419→        if len(steps) < 2:
   420→            k.drop_reason = "chain_too_short"
   421→            return k
   422→        # -- resource linkage: hidden fields that appear in a step's landing
   423→        # form but in no earlier form NOR in the step's own submitted fields —
   424→        # the object ids one step MINTS and a later step consumes. (The step's
   425→        # own fields are registered BEFORE its landing is scanned: the confirm
   426→        # form carries a hidden echo of the amount the create step submitted —
   427→        # that is data continuity, not resource production.)
   428→        known = {n for n, _ in entry_fields}
   429→        for s in steps:
   430→            known |= {n for n, _ in s.fields}
   431→            for form in self._in_cluster_forms(s.landing, s.action, k.cluster):
   432→                for f in form["fields"]:
   433→                    if isinstance(f, dict) and f.get("type") == "hidden" \
   434→                            and f["name"] not in known and not _is_token(f["name"]):
   435→                        k.resource_params[f["name"]] = f.get("value", "")
   436→        if not k.resource_params:
   437→            k.drop_reason = "no_resource_linkage"
   438→            return k
   439→        res_name = next(iter(k.resource_params))
   440→        # -- state oracle: the cluster GET page that echoes the resource value
   441→        origin = self._origin(members)
   442→        for path in sorted(members)[: self.cfg.max_pages]:
   443→            ep = members[path]
   444→            if (ep.method or "GET").upper() != "GET":
   445→                continue
   446→            r = self._get(engine, origin + path, "wf-oracle")
   447→            if not r.error and res_name and any(
   448→                    v and v in (r.body or "") for v in k.resource_params.values()):
   449→                k.oracle_url = ep.url
   450→                break
   451→        if not k.oracle_url:
   452→            k.drop_reason = "no_state_oracle"
   453→            return k
   454→        # -- execution markers: walk the chain AGAIN on the same session,
   455→        # snapshotting the oracle AFTER EVERY STEP, LIVE (a post-hoc snapshot
   456→        # would show the final state everywhere and the delta would be
   457→        # invisible — live-caught: the first v3.0.0 run dropped both flows as
   458→        # no_execution_marker). The producer step is the one whose landing
   459→        # CARRIES the fresh resource id; its snapshot's line is the
   460→        # pre-execution form, the final step's snapshot the executed form.
   461→        snaps: list[str] = []
   462→        steps2, _, _, _ = self._walk(members, engine,
   463→                                     after_step=lambda s, r: snaps.append(
   464→                                         self._get(engine, k.oracle_url,
   465→                                                   "wf-marker").body or ""))
   466→        if not steps2 or not snaps:
   467→            k.drop_reason = "marker_walk_failed"
   468→            return k
   469→        res_val2 = ""
   470→        for i, s in enumerate(steps2):
   471→            for form in self._in_cluster_forms(s.landing, s.action, k.cluster):
   472→                for f in form["fields"]:
   473→                    if isinstance(f, dict) and f.get("name") == res_name \
   474→                            and f.get("value"):
   475→                        res_val2, k.producer_idx = f["value"], i
   476→        if not res_val2 or k.producer_idx < 0 or k.producer_idx >= len(snaps):
   477→            k.drop_reason = "marker_walk_failed"
   478→            return k
   479→        la = (_lines_with(snaps[k.producer_idx], res_val2) or [""])[-1]
   480→        lb = (_lines_with(snaps[-1], res_val2) or [""])[-1]
   481→        added, removed = _word_delta(la, lb)
   482→        k.exec_added = [w for w in added if len(w) > 2]
   483→        k.exec_removed = [w for w in removed if len(w) > 2]
   484→        if not k.exec_added:
   485→            k.drop_reason = "no_execution_marker"
   486→            return k
   487→        # -- privilege-declared options across every form the chain exposes
   488→        seen_opts: list[tuple[str, str, str]] = []
   489→        for s in [WalkStep(action=entry, landing="")] + steps:
   490→            for form in self._in_cluster_forms(s.landing, s.action, k.cluster):
   491→                for triple in _privilege_options([form]):
   492→                    if triple not in seen_opts:
   493→                        seen_opts.append(triple)
   494→        k.priv_options = sorted(seen_opts)[: self.cfg.max_options]
   495→        return k
   496→
   497→    # -------------------------------------------------- oracle helpers
   498→    def _walk_resource(self, steps, res_name: str, cluster: str) -> str:
   499→        """The resource value a walk's OWN chain minted: the LAST value the
   500→        resource param takes across the walk's landing forms. Every walk mints
   501→        a fresh id — searching the LEARN-time examples instead (the first
   502→        v3.0.0 bug of this class) finds nothing and starves the privilege
   503→        baseline + attack."""
   504→        val = ""
   505→        for s in steps or []:
   506→            for form in self._in_cluster_forms(s.landing, s.action, cluster):
   507→                for f in form["fields"]:
   508→                    if isinstance(f, dict) and f.get("name") == res_name \
   509→                            and f.get("value"):
   510→                        val = f["value"]
   511→        return val

   533→        """A brand-new logged-in engine (isolated cookie jar + fresh app
   534→        state). None when the login fails (honest skip upstream)."""

   541→    def _atk_bypass(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   542→        """1. STEP BYPASS — submit the producer step directly on a session that
   543→        never ran its predecessor. Verified iff the oracle gains a NEW
   544→        canary-bearing object line (the object was created anyway)."""
   545→        if k.producer_idx < 1:
   546→            return None, {"verdict": "skipped", "note": "producer is the first step"}
   547→        producer = k.steps[k.producer_idx]
   548→        fields = self._token_refresh(producer.fields, entry_fields)
   549→        pre = self._oracle(engine, k, "wf-atk-bypass-pre")
   550→        r = self._post(engine, producer.action, fields, "wf-atk-bypass")
   551→        post = self._oracle(engine, k, "wf-atk-bypass-post")
   552→        new = [l for l in _lines_with(post, CANARY_CREATE)
   553→               if l not in set(_lines_with(pre, CANARY_CREATE))]
   554→        if not (r.ok and new):
   555→            return None, {"verdict": "no_proof", "status": r.status,
   556→                          "note": "no new canary-bearing object in the state oracle"}
   557→        param = next((n for n, _ in fields if not _is_token(n)), "")
   558→        f = self._finding(
   559→            "business_logic_step_bypass", producer.action, param,
   560→            Evidence("mandatory_step_bypassed",
   561→                     f"step {self._path_of(producer.action)} created an object on a "
   562→                     f"session that never ran its predecessor "
   563→                     f"{self._path_of(k.steps[k.producer_idx - 1].action)} — oracle "
   564→                     f"gained the line '{new[0].strip()[:120]}' — the mandatory "
   565→                     "earlier step is not enforced server-side", r.request_id),
   566→            f"state oracle gained a canary object without the predecessor step: "
   567→            f"'{new[0].strip()[:100]}'")
   568→        return f, {"verdict": "verified", "status": r.status,
   569→                   "line": new[0].strip()[:120]}
   570→
   571→    def _atk_out_of_order(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   572→        """2. OUT-OF-ORDER — the FINAL step with a never-created resource id.
   573→        Verified iff the oracle shows that id EXECUTED."""
   574→        if not k.resource_params:
   575→            return None, {"verdict": "skipped", "note": "no resource linkage"}
   576→        final = k.steps[-1]
   577→        res_name = next(iter(k.resource_params))
   578→        fresh_val = _synth_fresh_resource(k.resource_params[res_name])
   579→        fields = self._token_refresh(final.fields, entry_fields)
   580→        fields = [(n, fresh_val if n == res_name
   581→                   else (CANARY_TAMPER if _is_amount(n) else v)) for n, v in fields]
   582→        self._oracle(engine, k, "wf-atk-order-pre")
   583→        r = self._post(engine, final.action, fields, "wf-atk-order")
   584→        post = self._oracle(engine, k, "wf-atk-order-post")
   585→        lines = _lines_with(post, fresh_val)
   586→        if not (r.ok and lines and self._executed(lines[-1], k)):
   587→            return None, {"verdict": "no_proof", "status": r.status,
   588→                          "note": f"synthetic resource {fresh_val} not executed"}
   589→        f = self._finding(
   590→            "business_logic_out_of_order", final.action, res_name,
   591→            Evidence("final_step_without_producer",
   592→                     f"the final step {self._path_of(final.action)} EXECUTED a "
   593→                     f"never-created resource ({res_name}={fresh_val}) straight from "
   594→                     f"POSTed parameters — oracle line '{lines[-1].strip()[:120]}' — "
   595→                     "implicit creation + execution: the producing step was skipped",
   596→                     r.request_id),
   597→            f"oracle shows the synthetic resource executed: '{lines[-1].strip()[:100]}'")
   598→        return f, {"verdict": "verified", "status": r.status, "resource": fresh_val}
   599→
   600→    def _atk_tamper(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   601→        """3. VALUE TAMPER — walk the chain legitimately up to the final step,
   602→        then submit the final step with a DIFFERENT amount. Verified iff the
   603→        executed oracle line carries the tampered canary and not the approved
   604→        one."""
   605→        if len(k.steps) < 2 or k.producer_idx < 0:
   606→            return None, {"verdict": "skipped", "note": "chain too short"}
   607→        pre = self._oracle(engine, k, "wf-atk-tamper-pre")
   608→        # live walk up to (excluding) the final step; the pending form is the
   609→        # final step's form with the FRESH resource id and approved canary
   610→        steps, _, pending, _ = self._walk(k.members, engine,
   611→                                          max_chain=len(k.steps) - 1)
   612→        if not steps or pending is None:
   613→            return None, {"verdict": "no_proof", "note": "walk to final step failed"}
   614→        res_name = next(iter(k.resource_params))
   615→        res_val = ""
   616→        for f in pending["fields"]:
   617→            if f.get("name") == res_name:
   618→                res_val = f.get("value", "")
   619→        if not res_val:
   620→            return None, {"verdict": "no_proof",
   621→                          "note": "fresh resource value not observable"}
   622→        mid = self._oracle(engine, k, "wf-atk-tamper-mid")
   623→        if res_val not in mid:
   624→            return None, {"verdict": "no_proof",
   625→                          "note": "created resource not visible in the oracle"}
   626→        # synthesize the final step's fields, then FORCE the amount to the
   627→        # tamper canary — the approved value rides a HIDDEN field, so plain
   628→        # synthesis would carry it verbatim; the tamper is precisely changing it
   629→        fields = [(n, CANARY_TAMPER if _is_amount(n) else v)
   630→                  for n, v in self._synth_fields(pending)]
   631→        r = self._post(engine, pending["action"], fields, "wf-atk-tamper")
   632→        post = self._oracle(engine, k, "wf-atk-tamper-post")
   633→        line = (_lines_with(post, res_val) or [""])[-1]
   634→        if not (r.ok and line and CANARY_TAMPER in line and CANARY_CREATE not in line
   635→                and self._executed(line, k)):
   636→            return None, {"verdict": "no_proof", "status": r.status,
   637→                          "note": "approved value not overridable/executed"}
   638→        param = next((n for n, _ in fields if _is_amount(n)), "")
   639→        f = self._finding(
   640→            "business_logic_value_tamper", pending["action"], param,
   641→            Evidence("post_approval_override",
   642→                     f"the final step executed {res_name}={res_val} with the TAMPERED "
   643→                     f"value ({CANARY_TAMPER}) replacing the approved one "
   644→                     f"({CANARY_CREATE}) — oracle line '{line.strip()[:120]}' — values "
   645→                     "are re-trusted at execution time instead of bound at approval",
   646→                     r.request_id),
   647→            f"executed oracle line carries the tampered value: '{line.strip()[:100]}'")
   648→        return f, {"verdict": "verified", "status": r.status,
   649→                   "line": line.strip()[:120]}
   650→
   651→    def _atk_duplicate(self, k: FlowKnowledge, engine, entry_fields) -> tuple[Finding | None, dict]:
   652→        """4. DUPLICATE EXECUTION — run the full chain, then replay the final
   653→        step once. Verified iff the same resource shows MORE oracle lines
   654→        after the replay than before it."""
   655→        if len(k.steps) < 2:
   656→            return None, {"verdict": "skipped", "note": "chain too short"}
   657→        steps, _, _, _ = self._walk(k.members, engine)
   658→        if not steps or len(steps) < 2:
   659→            return None, {"verdict": "no_proof", "note": "full walk failed"}
   660→        final = steps[-1]
   661→        res_name = next(iter(k.resource_params))
   662→        res_val = ""
   663→        for n, v in final.fields:
   664→            if n == res_name:
   665→                res_val = v
   666→        mid = self._oracle(engine, k, "wf-atk-dup-mid")
   667→        if not res_val or res_val not in mid:
   668→            return None, {"verdict": "no_proof",
   669→                          "note": "executed resource not observable in the oracle"}
   670→        r = self._post(engine, final.action, final.fields, "wf-atk-dup-replay")
   671→        post = self._oracle(engine, k, "wf-atk-dup-post")
   672→        n_mid, n_post = len(_lines_with(mid, res_val)), len(_lines_with(post, res_val))
   673→        if not (r.ok and n_post > n_mid):
   674→            return None, {"verdict": "no_proof", "status": r.status,
   675→                          "note": f"executions {n_mid} -> {n_post}"}
   676→        f = self._finding(
   677→            "business_logic_duplicate_execution", final.action, res_name,
   678→            Evidence("non_idempotent_replay",
   679→                     f"replaying the final step executed {res_name}={res_val} AGAIN: "
   680→                     f"oracle occurrences {n_mid} -> {n_post} — no one-shot consumption "
   681→                     "(missing idempotency guard)", r.request_id),
   682→            f"same resource executed twice: {n_mid} -> {n_post} oracle occurrences")
   683→        return f, {"verdict": "verified", "status": r.status,
   684→                   "executions": f"{n_mid}->{n_post}"}
   685→
   686→    def _atk_privilege(self, k: FlowKnowledge, low_engine, entry_fields,
   687→                       low_role: str) -> tuple[Finding | None, dict]:
   688→        """5. PRIVILEGE BOUND — walk the chain under a LOWER-privilege session
   689→        choosing the option the UI labels privileged. Verified iff that
   690→        session's oracle shows the option EXECUTED (baseline: the option is
   691→        functional for the primary role — a control broken for everyone is
   692→        never reported as an escalation)."""
   693→        if not k.priv_options:
   694→            return None, {"verdict": "skipped", "note": "no privilege-declared option"}
   695→        fname, opt, label = k.priv_options[0]
   696→        if not k.priv_baselines.get(opt):
   697→            return None, {"verdict": "skipped",
   698→                          "note": f"option {opt} not functional for the primary role"}
   699→        pre = self._oracle(low_engine, k, "wf-atk-priv-pre")
   700→        steps, _, _, _ = self._walk(k.members, low_engine,
   701→                                    option_overrides={fname: opt})
   702→        if not steps:
   703→            return None, {"verdict": "no_proof", "note": "low-privilege walk failed"}
   704→        mid = self._oracle(low_engine, k, "wf-atk-priv-post")
   705→        res_val = self._walk_resource(steps, next(iter(k.resource_params)), k.cluster)
   706→        if not res_val or res_val in pre or res_val not in mid:
   707→            return None, {"verdict": "no_proof",
   708→                          "note": "low-privilege walk produced no observable object"}
   709→        line = (_lines_with(mid, res_val) or [""])[-1]
   710→        if not (opt in line and self._executed(line, k)):
   711→            return None, {"verdict": "no_proof",
   712→                          "note": f"option {opt} not executed by role={low_role}"}
   713→        f = self._finding(
   714→            "business_logic_privilege_escalation", k.steps[-1].action, fname,
   715→            Evidence("declared_privilege_not_enforced",
   716→                     f"the UI declares option {fname}={opt} '{label}' as privileged, "
   717→                     f"yet a role={low_role} session EXECUTED it — oracle line "
   718→                     f"'{line.strip()[:120]}' — the declared boundary is not enforced "
   719→                     "server-side", ""),
   720→            f"low-privilege session executed the admin-declared option: "
   721→            f"'{line.strip()[:100]}'",
   722→            authz_context=f"attacker=role:{low_role} declared_privilege={opt}")
   723→        return f, {"verdict": "verified", "role": low_role, "option": opt}

   725→    # -------------------------------------------------- run
   726→    def run(self, endpoints: list, main_engine, scope, bus, auth_profile,
   727→            authz_sessions: dict | None = None, timeout: float = 12.0,
   728→            logic_active: bool = False) -> dict:
   729→        """Phase 58 entry point. Learning always runs (when a form session
   730→        exists); the attack battery only with logic_active=True."""
   731→        stats = {"enabled": True, "logic_active": logic_active,
   732→                 "clusters_tried": 0, "flows_learned": 0, "attacks": 0,
   733→                 "verified": 0, "skipped": [], "requests": 0}
   734→        findings: list[Finding] = []
   735→        flows: list[FlowKnowledge] = []
   736→        if auth_profile is None or getattr(auth_profile, "mode", "form") != "form":
   737→            stats.update(enabled=False, skipped=["no_form_session"])
   738→            return {"findings": findings, "stats": stats, "flows": flows}
   739→        self._main_engine = main_engine
   740→        login_urls = {getattr(auth_profile, "login_url", ""),
   741→                      getattr(auth_profile, "action_url", "")}
   742→        clusters = self._cluster_members(endpoints, login_urls)
   743→        ranked = sorted(((d, m) for d, m in clusters.items() if self._eligible(m)),
   744→                        key=lambda t: (-self._score(t[0], t[1])[0],
   745→                                       -self._score(t[0], t[1])[1], t[0]))
   746→        for directory, members in ranked[: self.cfg.max_clusters]:
   747→            stats["clusters_tried"] += 1
   748→            k = self.learn(directory, members, main_engine)
   749→            if k.drop_reason:
   750→                stats["skipped"].append(f"{directory}:{k.drop_reason}")
   751→                continue
   752→            # privilege baselines under the PRIMARY session: an option broken
   753→            # for everyone is a bug, not an escalation — never reported as one.
   754→            res_name = next(iter(k.resource_params), "")
   755→            for fname, opt, label in k.priv_options:
   756→                pre = self._oracle(main_engine, k, "wf-base-pre")
   757→                bsteps, _, _, _ = self._walk(members, main_engine,
   758→                                             option_overrides={fname: opt})
   759→                post = self._oracle(main_engine, k, "wf-base-post")
   760→                fresh = self._walk_resource(bsteps, res_name, directory)
   761→                line = (_lines_with(post, fresh) or [""])[-1] if fresh else ""
   762→                k.priv_baselines[opt] = bool(fresh and opt in line
   763→                                             and self._executed(line, k))
   764→            stats["flows_learned"] += 1
   765→            flows.append(k)
   766→        if not logic_active:
   767→            stats["requests"] = self.requests
   768→            return {"findings": findings, "stats": stats, "flows": flows}
   769→        # ---- attack battery (explicit opt-in): fresh session per manipulation.
   770→        # The budget is PER FLOW (the five invariants are a flow-sized battery —
   771→        # a global cap let the first flow consume the whole budget and starved
   772→        # the rest: live-caught in benchmark trial 3, duplicate+privilege FN).
   773→        for k in flows:
   774→            flow_attacks = 0
   775→            for name, atk in (("step_bypass", self._atk_bypass),
   776→                              ("out_of_order", self._atk_out_of_order),
   777→                              ("value_tamper", self._atk_tamper),
   778→                              ("duplicate", self._atk_duplicate)):
   779→                if flow_attacks >= self.cfg.max_flow_attacks:
   780→                    stats["skipped"].append(f"{k.cluster}:attack_budget")
   781→                    break
   782→                flow_attacks += 1
   783→                eng = self._fresh_session(scope, bus, auth_profile, timeout)
   784→                if eng is None:
   785→                    stats["skipped"].append(f"{k.cluster}:fresh_login_failed")
   786→                    continue
   787→                entry_fields = []
   788→                r = self._get(eng, k.entry, "wf-atk-entry")
   789→                if not r.error:
   790→                    forms = self._in_cluster_forms(r.body or "", k.entry, k.cluster)
   791→                    entry_fields = self._synth_fields(forms[0]) if forms else []
   792→                if not entry_fields:
   793→                    stats["skipped"].append(f"{k.cluster}:entry_form_lost")
   794→                    continue
   795→                f, detail = atk(k, eng, entry_fields)
   796→                detail.update(attack=name, flow=k.cluster)
   797→                self.log.append(detail)
   798→                stats["attacks"] += 1
   799→                if f:
   800→                    findings.append(f)
   801→                    stats["verified"] += 1
   802→            # privilege: needs a strictly lower-privilege logged-in session
   803→            lows = sorted((s for s in (authz_sessions or {}).values()
   804→                           if s.get("rank", 3) > 0),
   805→                          key=lambda s: s["account"].name)
   806→            if not lows:
   807→                stats["skipped"].append(f"{k.cluster}:no_lower_privilege_session")
   808→            else:
   809→                low = lows[0]
   810→                entry_fields = []
   811→                r = self._get(low["engine"], k.entry, "wf-atk-priv-entry")
   812→                if not r.error:
   813→                    forms = self._in_cluster_forms(r.body or "", k.entry, k.cluster)
   814→                    entry_fields = self._synth_fields(forms[0]) if forms else []
   815→                if entry_fields:
   816→                    f, detail = self._atk_privilege(k, low["engine"], entry_fields,
   817→                                                    low["account"].role)
   818→                    detail.update(attack="privilege", flow=k.cluster)
   819→                    self.log.append(detail)
   820→                    stats["attacks"] += 1
   821→                    if f:
   822→                        findings.append(f)
   823→                        stats["verified"] += 1
   824→                else:
   825→                    stats["skipped"].append(f"{k.cluster}:low_entry_form_lost")
   826→        stats["requests"] = self.requests
   827→        stats["attack_log"] = self.log
   828→        return {"findings": findings, "stats": stats, "flows": flows}
