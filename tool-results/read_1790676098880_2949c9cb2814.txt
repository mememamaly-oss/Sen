    16→class ScanConfig:
    17→    def __init__(self, allow_private=False, max_pages=25, crawl=True, active=True,
    18→                 passive=True, timeout=12.0, seed=None, strategy_path=None,
    19→                 allowed_domains=None, oast=None, browser=True, auth=None):

    21→        self.max_pages = max_pages
    22→        self.crawl, self.active, self.passive = crawl, active, passive
    23→        self.timeout, self.seed = timeout, seed
    24→        self.strategy_path = strategy_path
    25→        self.allowed_domains = allowed_domains or []
    26→        self.oast = oast          # OastServer | None — enables OOB (blind-XXE) confirmation
    27→        self.browser = browser    # Phase 50: DOM engine (graceful skip when unavailable)
    28→        self.auth = auth          # Phase 51: AuthProfile | None — authenticated scanning

    31→class SentinelScanner:
    32→    def __init__(self, target_url: str, cfg: ScanConfig | None = None):
    33→        self.cfg = cfg or ScanConfig()
    34→        self.run_id, self.seed = new_run_id(self.cfg.seed)
    35→        self.bus = EventBus(self.run_id)
    36→        self.target = Target(target_url)
    37→        self.scope = ScopeRule(allowed_domains=[self.target.host] + self.cfg.allowed_domains,
    38→                               allow_private=self.cfg.allow_private)
    39→        self.engine = RequestEngine(self.scope, self.bus, timeout=self.cfg.timeout)
    40→        self.verifier = VerificationEngine(self.engine, self.bus, oast=self.cfg.oast)
    41→        self.browser = None       # Phase 50: BrowserEngine (set in run() when enabled+available)
    42→        self.pipeline = FindingPipeline(self.bus)
    43→        self.strategies = StrategyStore(self.cfg.strategy_path)
    44→        self.failures = FailureMemory()
    45→        self.classifier = RepairClassifier()
    46→        self.findings: list[Finding] = []
    47→        self.stats = {"requests": 0, "endpoints": 0, "candidates": 0, "verified": 0, "rejected": 0,
    48→                      "duration": 0.0}
    49→
    50→    # ---------------------------------------------------------------- run
    51→    def run(self) -> dict:
    52→        t0 = time.time()
    53→        self.bus.emit("ScanStarted", target=self.target.url, run=self.run_id,
    54→                      scanner=SCANNER_VERSION, seed=self.seed)
    55→
    56→        # ---- Phase 51: authenticate BEFORE discovery — the session cookie (ENG-05
    57→        # jar) then rides every later request, and the authenticated landing page
    58→        # becomes a second crawl seed (the post-login surface is linked ONLY from
    59→        # post-login pages: /bank/* on Altoro is invisible from the anonymous home).
    60→        auth_seeds: list[str] = []
    61→        self.auth_engine = None
    62→        if self.cfg.auth is not None:
    63→            self.auth_engine = AuthEngine(self.cfg.auth, self.engine, self.bus)
    64→            ok = self.auth_engine.login()
    65→            self.stats["authenticated"] = ok
    66→            if ok:
    67→                # AUTH-03 watchdog proxy: from here on, session loss anywhere in
    68→                # the scan (discovery, detectors, verifier, browser) self-heals.
    69→                self.engine = AuthedEngineProxy(self.engine, self.auth_engine)
    70→                self.verifier.engine = self.engine
    71→                auth_seeds.append(self.auth_engine.state.landing_url)
    72→                # the authenticated crawl genuinely has more surface — grow budget
    73→                # deterministically (fixed 1.5x, no heuristics)
    74→                self.cfg.max_pages = int(self.cfg.max_pages * 1.5)
    75→            else:
    76→                self.bus.emit("ScanContinuesUnauthenticated",
    77→                              reason=self.auth_engine.state.last_error)
    78→
    79→        # ---- Phase 4: discovery (extra_seeds carry the authenticated landing;
    80→        # the login page itself is excluded from page-fetches — re-fetching it
    81→        # clobbers the authenticated session cookie with a fresh anonymous one)
    82→        d = DiscoveryEngine(self.engine, self.cfg.max_pages)
    83→        endpoints: list[Endpoint] = [Endpoint(url=self.target.url, source=DiscoverySource.SEED)]
    84→        if self.cfg.crawl:
    85→            found, js, crawl_stats = d.crawl(self.target.url, extra_seeds=auth_seeds,
    86→                                             exclude_urls=[self.cfg.auth.login_url] if self.cfg.auth else None)
    87→            endpoints.extend(e for e in found if e.id != endpoints[0].id)
    88→            robots = d.robots(self.target.url)
    89→            endpoints.extend(e for e in robots if all(e.id != x.id for x in endpoints))
    90→            self.stats["crawl"] = crawl_stats
    91→            self.bus.emit("DiscoveryCompleted", **crawl_stats, js_endpoints=len(js))
    92→        self.stats["endpoints"] = len(endpoints)
    93→
    94→        # ---- Phase 50: browser/DOM engine — render-merge BEFORE planning so active
    95→        # detectors cover runtime-discovered surfaces (JS-built forms, rendered links).
    96→        dom_sinks = list(getattr(d, "dom_sinks", [])) if self.cfg.crawl else []
    97→        if self.cfg.browser:
    98→            from .browser import BrowserEngine, browser_available
    99→            if browser_available():
   100→                self.browser = BrowserEngine(self.engine, timeout=self.cfg.timeout)
   101→                self.verifier.browser = self.browser
   102→                existing = {e.id: e for e in endpoints}
   103→                # Render budget: script-bearing pages first (runtime forms / DOM logic
   104→                # only exist where scripts run — live-caught: a naive first-N endpoint
   105→                # selection missed the JS-built form page entirely), then dom-sink
   106→                # pages, then remaining GET pages up to the cap.
   107→                seen_t = set()
   108→                render_targets = []
   109→                for u in (list(getattr(d, "script_pages", [])) +
   110→                          [s.url for s in dom_sinks] +
   111→                          [e.url for e in endpoints if (e.method or "GET") == "GET"]):
   112→                    if u and u not in seen_t:
   113→                        seen_t.add(u)
   114→                        render_targets.append(u)
   115→                render_targets = render_targets[:15]
   116→                new_eps = self.browser.render_merge(render_targets, self.target.host, existing)
   117→                endpoints.extend(e for e in new_eps if e.id not in {x.id for x in endpoints})
   118→                self.stats["endpoints"] = len(endpoints)
   119→                self.stats["browser_pages"] = self.browser.pages_rendered
   120→                self.bus.emit("BrowserRenderCompleted", pages=self.browser.pages_rendered,
   121→                              new_endpoints=len(new_eps), dom_sink_candidates=len(dom_sinks))
   122→            else:
   123→                self.bus.emit("BrowserUnavailable",
   124→                              note="playwright/chromium not installed — static fallback active")
   125→
   126→        # ---- Phase 7: plan; Phase 8: detect
   127→        detectors = {d.name: d(self.engine, self.bus, self.strategies) for d in ACTIVE_DETECTORS}
   128→        # Phase 53: OAST-channel detectors (SSRF / blind command injection) receive
   129→        # the armed channel; without one they stay inert (response oracles for these
   130→        # classes are noise — live-observed on the cycling stock endpoint).
   131→        for _d in detectors.values():
   132→            if getattr(_d, "needs_oast", False):
   133→                _d.oast = self.cfg.oast
   134→        vuln_to_detector = {"sql_injection": "sqli", "xss_reflected": "xss",
   135→                            "path_traversal": "traversal", "open_redirect": "redirect", "ssrf": "ssrf",
   136→                            "header_injection": "crlf", "xxe": "xxe",
   137→                            "command_injection": "cmdi", "cors_misconfig": "cors"}
   138→        plan = []
   139→        for ep in endpoints:
   140→            plan.extend(plan_for(ep))
   141→        self.bus.emit("ScanPlanReady", planned=len(plan))
   142→
   143→        for ep in endpoints:
   144→            if self.cfg.passive:
   145→                r = self.engine.send(RequestMutation(endpoint=ep, note="passive-baseline"))
   146→                if not r.error:
   147→                    for f in passive_audit(ep, r):
   148→                        self.findings.append(self.verifier.verify(f, ep))
   149→            if not self.cfg.active:
   150→                continue
   151→            for dname, det in detectors.items():
   152→                try:
   153→                    if dname == "sqli":
   154→                        cands = det.candidates(ep)
   155→                        if not cands: continue
   156→                        # PER-PARAM ANALYSIS (live-caught on doLogin: old flat resp_map kept
   157→                        # only the LAST param's responses, silently discarding earlier params)
   158→                        from itertools import groupby
   159→                        for pname, grp in groupby(cands, key=lambda t: t[1].param_name):
   160→                            grp = list(grp)
   161→                            resp_map = {}
   162→                            for strat, m in grp:
   163→                                r = self.engine.send(m)
   164→                                if r.error:
   165→                                    fid = self.failures.record("sqli", r.error, {"url": ep.url})
   166→                                    self.bus.emit("ScannerFailed", scanner="sqli", failure=fid, error=r.error)
   167→                                    continue
   168→                                resp_map[strat] = r
   169→                            param_obj = next((p for p in ep.parameters if p.name == pname), None)
   170→                            if param_obj is None or not resp_map:
   171→                                continue
   172→                            cand = det.analyze((ep, param_obj), resp_map)
   173→                            if cand:
   174→                                cand.run_id = self.run_id
   175→                                self.stats["candidates"] += 1
   176→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   177→                                out_f = self.verifier.verify(cand, ep)
   178→                                self.findings.append(out_f)
   179→                                # Phase 19/25: record VERIFIED outcomes only into strategy store
   180→                                for st in ("quote_error", "boolean_and", "or_redirect", "or_differential", "quote_status"):
   181→                                    self.strategies.record("sqli", st, out_f.status == FindingStatus.VERIFIED,
   182→                                                           context=self.target.host)
   183→                    elif dname == "xxe":
   184→                        # DETECT-08 two-phase flow: candidates() returns recon probes (XML
   185→                        # with form fields, root learned from JS); the recon response names
   186→                        # missing elements ("Store ID could not be parsed"), expand() builds
   187→                        # the entity differential probes, analyze() compares them.
   188→                        cands = det.candidates(ep)
   189→                        for strat, m in cands:
   190→                            r0 = self.engine.send(m)
   191→                            if r0.error or r0.status == 0:
   192→                                continue
   193→                            root = strat.split("|")[1]
   194→                            probes = det.expand(ep, r0, root)
   195→                            if not probes:
   196→                                continue
   197→                            resp_map = {}
   198→                            for s2, m2 in probes:
   199→                                r = self.engine.send(m2)
   200→                                if r.error:
   201→                                    fid = self.failures.record("xxe", r.error, {"url": ep.url})
   202→                                    self.bus.emit("ScannerFailed", scanner="xxe", failure=fid, error=r.error)
   203→                                    continue
   204→                                resp_map[s2] = r
   205→                            if not resp_map:
   206→                                continue
   207→                            cand = det.analyze((ep, None), resp_map)
   208→                            # Phase 49 — OOB fallback: response oracle impossible (noise /
   209→                            # non-deterministic answers) but an OAST listener is armed ->
   210→                            # the target itself becomes the evidence channel (blind XXE).
   211→                            if cand is None and self.cfg.oast is not None:
   212→                                self.bus.emit("OobAttemptStarted", scanner="xxe", url=ep.url)
   213→                                cand = det.oob_attempt(ep, root, r0, self.cfg.oast)
   214→                            if cand:
   215→                                cand.run_id = self.run_id
   216→                                self.stats["candidates"] += 1
   217→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   218→                                out_f = self.verifier.verify(cand, ep)
   219→                                self.findings.append(out_f)
   220→                                strat_name = ("oob_callback" if any(e.kind == "oob_callback" for e in cand.evidence)
   221→                                              else "entity_differential")
   222→                                self.strategies.record("xxe", strat_name,
   223→                                                       out_f.status == FindingStatus.VERIFIED,
   224→                                                       context=self.target.host)
   225→                    elif dname == "cors":
   226→                        # DETECT-13: header-only probe (Origin reflection + credentials) —
   227→                        # has no param, so it cannot ride the param-bound generic loop.
   228→                        cand = det.run(ep)
   229→                        if cand:
   230→                            cand.run_id = self.run_id
   231→                            self.stats["candidates"] += 1
   232→                            self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   233→                            out_f = self.verifier.verify(cand, ep)
   234→                            self.findings.append(out_f)
   235→                            self.strategies.record("cors", "primary",
   236→                                                   out_f.status == FindingStatus.VERIFIED,
   237→                                                   context=self.target.host)
   238→                    else:
   239→                        for m in det.candidates(ep):
   240→                            if isinstance(m, tuple):
   241→                                m = m[1]
   242→                            r = self.engine.send(m)
   243→                            if r.error:
   244→                                fid = self.failures.record(det.name, r.error, {"url": ep.url})
   245→                                self.bus.emit("ScannerFailed", scanner=det.name, failure=fid, error=r.error)
   246→                                continue
   247→                            pair = (ep, next((p for p in ep.parameters if p.name == m.param_name), ep.parameters[0] if ep.parameters else None))
   248→                            if not pair[1]: continue
   249→                            cand = det.analyze(pair, r)
   250→                            if cand:
   251→                                cand.run_id = self.run_id
   252→                                self.stats["candidates"] += 1
   253→                                self.bus.emit("CandidateCreated", type=cand.type, url=ep.url, param=cand.parameter)
   254→                                out_f = self.verifier.verify(cand, ep)
   255→                                self.findings.append(out_f)
   256→                                self.strategies.record(dname, "primary", out_f.status == FindingStatus.VERIFIED,
   257→                                                       context=self.target.host)
   258→                except Exception as e:
   259→                    fid = self.failures.record(dname, f"{type(e).__name__}: {e}", {"url": ep.url})
   260→                    self.bus.emit("ScannerFailed", scanner=dname, failure=fid, error=str(e)[:200])
   261→
   262→        # ---- Phase 50: DOM XSS execution probes (browser; static sink candidates).
   263→        # search sources probe the extracted param names; hash sources take the
   264→        # payload directly in the fragment; referrer/name sources are a documented
   265→        # limit (cannot be set by plain navigation).
   266→        if self.browser is not None and self.cfg.active and dom_sinks:
   267→            try:
   268→                import hashlib as _hb
   269→                seen_probe = set()
   270→                for sink in dom_sinks:
   271→                    if sink.source == "search":
   272→                        probe_params = [p for p in sink.params[:3] if p]
   273→                    elif sink.source == "hash":
   274→                        probe_params = [""]
   275→                    else:
   276→                        continue
   277→                    for pname in probe_params:
   278→                        pkey = f"{sink.url}|{pname}|{sink.source}"
   279→                        if pkey in seen_probe:
   280→                            continue
   281→                        seen_probe.add(pkey)
   282→                        canary = "z" + _hb.sha1(pkey.encode()).hexdigest()[:9]
   283→                        # canonical page URL (crawl params stripped) — the finding and
   284→                        # its re-probe must be reproducible from a clean URL
   285→                        page_url = sink.url.split("?")[0].split("#")[0]
   286→                        r = self.browser.probe_dom_xss(page_url, pname, canary, source=sink.source)
   287→                        if r and r.get("fired"):
   288→                            cand = Finding(type="xss_dom", severity=Severity.HIGH,
   289→                                           endpoint_url=page_url, parameter=pname, method="GET",
   290→                                           evidence=[Evidence("dom_execution",
   291→                                                      f"source={sink.source} sink={sink.sink} — {r['detail']}; "
   292→                                                      f"static flow: {sink.snippet}")],
   293→                                           status=FindingStatus.CANDIDATE)
   294→                            cand.run_id = self.run_id
   295→                            self.stats["candidates"] += 1
   296→                            self.bus.emit("CandidateCreated", type=cand.type, url=sink.url, param=pname)
   297→                            out_f = self.verifier.verify(cand, None)
   298→                            self.findings.append(out_f)
   299→                            self.strategies.record("dom_xss", sink.sink,
   300→                                                   out_f.status == FindingStatus.VERIFIED,
   301→                                                   context=self.target.host)
   302→                self.stats["browser_probes"] = self.browser.probes_sent
   303→            except Exception as e:
   304→                fid = self.failures.record("dom_xss", f"{type(e).__name__}: {e}", {})
   305→                self.bus.emit("ScannerFailed", scanner="dom_xss", failure=fid, error=str(e)[:200])
   306→        if self.browser is not None:
   307→            try:
   308→                self.browser.close()
   309→            except Exception:
   310→                pass
   311→
   312→        # ---- Phase 12: dedup + correlation + risk
   313→        verified = [f for f in self.findings if f.status == FindingStatus.VERIFIED]
   314→        rejected = [f for f in self.findings if f.status == FindingStatus.REJECTED]
   315→        self.stats.update({"requests": self.engine.requests_sent,
   316→                           "verified": len(verified), "rejected": len(rejected),
   317→                           "duration": round(time.time() - t0, 2)})
   318→        if self.auth_engine is not None:
   319→            self.stats["auth"] = {
   320→                "authenticated": self.auth_engine.state.authenticated,
   321→                "landing": self.auth_engine.state.landing_url,
   322→                "logins": self.auth_engine.state.logins,
   323→                "reauths": self.auth_engine.state.reauths,
   324→                "failed_reauths": self.auth_engine.state.failed_reauths,
   325→            }
   326→        # Phase 52: OAST channel transparency — which channel served OOB evidence
   327→        # (local loopback listener vs public webhook provider) and how many
   328→        # interactions it recorded. Blind-vuln confirmations are auditable from
   329→        # the report alone, without re-running the scan.
   330→        if self.cfg.oast is not None:
   331→            try:
   332→                self.stats["oast"] = {"channel": getattr(self.cfg.oast, "channel", "local"),
   333→                                      "interactions": len(self.cfg.oast.hits())}
   334→            except Exception:
   335→                self.stats["oast"] = {"channel": getattr(self.cfg.oast, "channel", "local"),
   336→                                      "interactions": -1}
   337→        ded = self.pipeline.dedup(verified)
   338→        final = self.pipeline.correlate(ded)
   339→        for f in final:
   340→            f.confidence = max(f.confidence, 0.5)
   341→        self.findings = final + rejected
   342→        for f in final:
   343→            self.bus.emit("FindingCreated", id=f.id, type=f.type, url=f.endpoint_url,
   344→                          param=f.parameter, risk=self.pipeline.risk(f))
   345→        self.bus.emit("ScanCompleted", **self.stats)
   346→        return self.report()
   347→
   348→    def report(self) -> dict:
   349→        return {
   350→            "run_id": self.run_id, "seed": self.seed, "scanner": SCANNER_VERSION,
   351→            "target": self.target.url, "stats": self.stats,
   352→            "findings": [f.to_dict() | {"risk": self.pipeline.risk(f)} for f in self.findings
   353→                         if f.status == FindingStatus.VERIFIED],
   354→            "rejected_candidates": [f.to_dict() for f in self.findings
   355→                                    if f.status == FindingStatus.REJECTED],
   356→        }

   359→# ============================================================ DOCTOR (Phase 44/45)
   360→def doctor(allow_private=False) -> dict:
   361→    import socket, ssl, subprocess, sys, platform
   362→    checks = []
   363→
   364→    def add(name, ok, detail, fix=""):
   365→        checks.append({"check": name, "status": "PASS" if ok else ("WARN" if detail.startswith("warn") else "FAIL"),
   366→                       "detail": detail, "auto_fix": fix if not ok else ""})
   367→
   368→    add("runtime", True, f"Python {sys.version.split()[0]} on {platform.system()}")
   369→    # DNS
   370→    try:
   371→        socket.gethostbyname("scanme.nmap.org")
   372→        add("dns_resolution", True, "resolved scanme.nmap.org")
   373→    except Exception as e:
   374→        add("dns_resolution", False, str(e)[:80], "check /etc/resolv.conf")
   375→    # outbound TCP
   376→    try:
   377→        s = socket.create_connection(("scanme.nmap.org", 80), timeout=8); s.close()
   378→        add("network_egress", True, "TCP connect scanme.nmap.org:80 OK")
   379→    except Exception as e:
   380→        add("network_egress", False, str(e)[:80])
   381→    # TLS
   382→    try:
   383→        ctx = ssl.create_default_context(); ctx.check_hostname = False; ctx.verify_mode = ssl.CERT_NONE
   384→        add("tls_stack", True, OpenSSL_ver := ssl.OPENSSL_VERSION)
   385→    except Exception as e:
   386→        add("tls_stack", False, str(e)[:80])
   387→    # scope policy sanity
   388→    from .core import is_private_ip
   389→    add("scope_policy", is_private_ip("127.0.0.1") and is_private_ip("10.0.0.5") and not is_private_ip("45.33.32.156"),
   390→        "private ranges blocked, public allowed" + (" (+private override)" if allow_private else ""))
   391→    # detector registry
   392→    add("detector_registry", len(ACTIVE_DETECTORS) >= 4, f"{len(ACTIVE_DETECTORS)} active detectors registered")
   393→    # strategy store
   394→    add("strategy_store", True, "verified-outcome Wilson ordering enabled")
   395→    # persistence layer (Phase 46): sqlite3 available + schema initializes + round-trip
   396→    try:
   397→        import tempfile, os as _os, sqlite3
   398→        from .store import Store
   399→        p = _os.path.join(tempfile.mkdtemp(), "doctor.db")
   400→        st = Store(p)
   401→        st.save_run({"run_id": "run-doctor", "seed": "s", "scanner": "doctor",
   402→                     "target": "http://doctor.local", "stats": {"requests": 1},
   403→                     "findings": [{"id": "F-d", "type": "missing_security_headers",
   404→                                   "severity": "low", "status": "verified",
   405→                                   "url": "http://doctor.local/", "parameter": "", "method": "GET",
   406→                                   "confidence": 0.98, "risk": 20, "evidence": [], "verification": {}}],
   407→                     "rejected_candidates": []})
   408→        ok = len(st.list_findings()) == 1 and st.stats_summary()["runs"] == 1
   409→        add("persistence_layer", ok,
   410→            f"SQLite store OK (schema + write + query round-trip) at {sqlite3.sqlite_version}")
   411→    except Exception as e:
   412→        add("persistence_layer", False, str(e)[:80])
   413→    # repair registry
