   486→RISK_WEIGHTS = {"sql_injection": 9, "xss_reflected": 7, "xss_dom": 7, "path_traversal": 7, "xxe": 9,
   487→                "command_injection": 9, "ssrf": 7, "cors_misconfig": 6,

   494→                "tech_banner_disclosure": 1, "vulnerable_js_dependency": 2}

   496→class FindingPipeline:
   497→    def __init__(self, bus: EventBus):
   498→        self.bus = bus
   499→
   500→    def dedup(self, findings: list[Finding]) -> list[Finding]:
   501→        seen: dict[str, Finding] = {}
   502→        for f in findings:
   503→            fp = f.fingerprint()
   504→            if fp in seen:
   505→                seen[fp].evidence.extend(f.evidence)
   506→                self.bus.emit("FindingDeduplicated", fingerprint=fp, kept=seen[fp].id, dropped=f.id)
   507→            else:
   508→                seen[fp] = f
   509→        return list(seen.values())
   510→
   511→    def correlate(self, findings: list[Finding]) -> list[Finding]:
   512→        """Merge duplicate evidence for the SAME injection point (type + full path + param).
   513→        Different endpoints/parameters stay distinct findings — they are distinct root causes."""
   514→        roots: dict[str, Finding] = {}
   515→        for f in sorted(findings, key=lambda x: -RISK_WEIGHTS.get(x.type, 1)):
   516→            key = f"{f.type}|{f.endpoint_url.split('?')[0]}|{f.parameter}"
   517→            if key in roots:
   518→                roots[key].evidence.extend(f.evidence[:2])
   519→            else:
   520→                roots[key] = f
   521→        return list(roots.values())
   522→
   523→    def risk(self, f: Finding) -> int:
   524→        """technical severity x confidence x exposure, 0-100."""
   525→        base = RISK_WEIGHTS.get(f.type, 3) * 10
   526→        conf = f.confidence or 0.5
   527→        auth_factor = 1.0
   528→        return min(100, int(base * conf * auth_factor))

   531→# ============================================================ STRATEGY STORE (Phase 19/25/26)
   532→def wilson_lower(successes: int, n: int, z: float = 1.96) -> float:
   533→    """Wilson score lower bound — sample-size-aware (spec 26)."""
   534→    if n == 0: return 0.0
   535→    p = successes / n
   536→    denom = 1 + z * z / n
   537→    centre = p + z * z / (2 * n)
   538→    margin = z * math.sqrt((p * (1 - p) + z * z / (4 * n)) / n)
   539→    return max(0.0, (centre - margin) / denom)

   542→class StrategyStore:
   543→    """Learning = deterministic statistics from VERIFIED outcomes only (spec 51)."""
   544→    def __init__(self, path: str | None = None):
   545→        self.path = path
   546→        self.stats: dict[str, dict] = {}
   547→        if path and os.path.exists(path):
   548→            self.stats = json.load(open(path))
   549→
   550→    def key(self, scanner: str, strategy: str, context: str = "default") -> str:
   551→        return f"{scanner}:{context}:{strategy}"
   552→
   553→    def record(self, scanner: str, strategy: str, verified: bool, context: str = "default"):
   554→        k = self.key(scanner, strategy, context)
   555→        s = self.stats.setdefault(k, {"n": 0, "verified": 0, "failed": 0})
   556→        s["n"] += 1
   557→        s["verified" if verified else "failed"] += 1
   558→        self.flush()
   559→
   560→    def order(self, scanner: str, strategies: list[str], context: str = "default") -> list[str]:
   561→        """Deterministic ordering by Wilson lower bound; ties broken alphabetically."""
   562→        def lb(st):
   563→            s = self.stats.get(self.key(scanner, st, context), {"n": 0, "verified": 0})
   564→            return (wilson_lower(s["verified"], s["n"]), s["n"])
   565→        return sorted(strategies, key=lambda st: (-lb(st)[0], st))
   566→
   567→    def flush(self):
   568→        if self.path:
   569→            json.dump(self.stats, open(self.path, "w"), indent=1)

   622→class RepairGate:
   623→    """spec 23/38: repair candidates are applied in an isolated copy, tested, benchmarked,
   624→    compared, then accepted or rolled back. Never direct self-modification."""
   625→    def __init__(self, bench_fn, regression_fn):
   626→        self.bench_fn, self.regression_fn = bench_fn, regression_fn
   627→
   628→    def attempt(self, failure_id: str, rule_name: str, action: dict, apply_fn, label: str = "") -> dict:
   629→        result = {"failure_id": failure_id, "rule": rule_name, "label": label,
   630→                  "steps": [], "accepted": False}
   631→        for step, fn, ok_check in [
   632→            ("apply_in_isolation", apply_fn, lambda r: r is True),
   633→            ("regression_suite", self.regression_fn, lambda r: r.get("passed") is True),
   634→            ("benchmark_compare", self.bench_fn, lambda r: r.get("acceptable") is True),
   635→        ]:
   636→            try:
   637→                r = fn()
   638→                ok = ok_check(r)
   639→            except Exception as e:
   640→                r, ok = {"error": str(e)}, False
   641→            result["steps"].append({"step": step, "ok": ok, "detail": r if isinstance(r, dict) else r})
   642→            if not ok:
   643→                result["accepted"] = False
   644→                result["rollback_reason"] = f"gate failed at: {step}"
   645→                return result
   646→        result["accepted"] = True
   647→        return result
